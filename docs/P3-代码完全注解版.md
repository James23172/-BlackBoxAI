## 1.1 Point.java — 坐标点

```java
package inspection.common.model;                       // ① 包声明：组织代码的命名空间

import java.util.Objects;                              // ② 引入 Objects 工具类（用于 hashCode）

/**
 * 坐标点类
 * 表示地图上的一个二维坐标 (x, y)
 * x 从左到右递增，y 从上到下递增
 */
public class Point {                                    // ③ 类声明：public 表示任何包都可访问

    /** 横坐标（列号），从左到右 [0, mapWidth-1] */
    public int x;                                       // ④ 公开字段（项目风格选择：字段公开而非 getter）

    /** 纵坐标（行号），从上到下 [0, mapHeight-1] */
    public int y;

    /**
     * 默认构造函数（无参）
     * ← 存在原因：fastjson2 反序列化 JSON 时需要无参构造器来创建对象
     *    然后通过反射设置字段值
     */
    public Point() {}

    /**
     * 全参构造函数
     * @param x 横坐标
     * @param y 纵坐标
     */
    public Point(int x, int y) {
        this.x = x;                                     // ⑤ this.x = 参数 x；左边是字段，右边是参数
        this.y = y;
    }

    // ===== getter/setter =====（供 fastjson2 序列化/反序列化使用）
    public int getX() { return x; }
    public int getY() { return y; }
    public void setX(int x) { this.x = x; }
    public void setY(int y) { this.y = y; }

    /**
     * 计算曼哈顿距离
     * 网格地图不能斜走，所以距离 = |Δx| + |Δy|
     *
     * 示例：
     *   (0,0).distanceTo((3,4)) = |3-0| + |4-0| = 7
     *
     * 用途：A* 算法的启发函数、TargetPlanner 选最近未探索格
     *
     * @param other 另一个坐标点
     * @return 曼哈顿距离（非负整数）
     */
    public int distanceTo(Point other) {
        return Math.abs(this.x - other.x) + Math.abs(this.y - other.y);
        //     ↑ Math.abs() 返回绝对值，如 Math.abs(-5) = 5
    }

    /**
     * 判断两个坐标点是否相等
     * ← @Override 表示重写父类 Object 的方法
     * ← 重写原因：HashSet<Point> 和 HashMap 依赖 equals() 判断元素是否相同
     *   如果不重写，两个 new Point(10,10) 会被认为是不同的对象
     *
     * @param o 要比较的对象
     * @return x 和 y 都相等时返回 true
     */
    @Override
    public boolean equals(Object o) {
        if (this == o) return true;                     // ⑥ 如果是同一个对象引用 → 直接返回 true
        if (!(o instanceof Point)) return false;        // ⑦ 如果不是 Point 类型 → 返回 false
        Point p = (Point) o;                            // ⑧ 类型转换（安全的，因为已通过 instanceof 检查）
        return x == p.x && y == p.y;                    // ⑨ 两个坐标值都相等才算相等
    }

    /**
     * 计算哈希码（HashSet / HashMap 用）
     * ← 规则：如果 a.equals(b) == true，则 a.hashCode() == b.hashCode()
     *   反之不一定成立（哈希碰撞）
     *
     * @return 基于 x 和 y 的哈希码
     */
    @Override
    public int hashCode() {
        return Objects.hash(x, y);                      // ⑩ Objects.hash() 自动组合多个字段计算哈希
    }

    /**
     * 转换为字符串表示
     * 格式："(x,y)"
     * 用途：日志输出、调试
     */
    @Override
    public String toString() {
        return "(" + x + "," + y + ")";
    }
}
```

## 1.2 CarStatus.java — 小车状态枚举

```java
package inspection.common.enums;

/**
 * 小车三状态枚举
 *
 * 状态流转（有限状态机 FSM）：
 *   IDLE ──(收到 MOVE_STEP 命令)──→ MOVING
 *   MOVING ──(路径走完/第二步前瞻发现障碍)──→ IDLE
 *   MOVING ──(下一步有障碍物)──→ BLOCKED
 *   BLOCKED ──(超时 200ms 后 Controller 恢复)──→ IDLE
 */
public enum CarStatus {                                 // ① enum 关键字创建枚举类型（值的固定集合）

    /** 空闲状态：等待 Controller 分配路径和移动命令 */
    IDLE,                                               // ② 枚举常量（值是 CarStatus.IDLE）

    /** 移动中：正在执行移动操作 */
    MOVING,

    /** 受阻状态：前方有障碍物，等待超时后 Controller 重新规划路径 */
    BLOCKED

    // ← 枚举自动提供 .name() 方法返回字符串 "IDLE"/"MOVING"/"BLOCKED"
    // ← 枚举自动提供 .valueOf("IDLE") 方法将字符串转回枚举值
    // ← 这两者被 BlackboardClient 读写 Redis 时使用：
    //     写入：jedis.set(key, status.name())  → Redis 存 "IDLE"
    //     读取：CarStatus.valueOf(val)          → Redis "IDLE" → CarStatus.IDLE
}
```

## 1.4 MQMessage.java — 消息信封

```java
package inspection.common.model;

import java.util.Map;

/**
 * RabbitMQ 消息模型
 * 所有模块间的通信都使用此格式
 *
 * JSON 格式示例：
 * {
 *   "cmd": "NAVIGATE",
 *   "data": {"carId": "Car001"},
 *   "timestamp": 1717401600000
 * }
 */
public class MQMessage {

    /** 命令类型：如 "NAVIGATE", "MOVE_STEP" */
    public String cmd;                                  // ① 存储 CommandType.name() 的值

    /** 命令携带的数据：键值对，具体内容随命令类型不同而变化 */
    public Map<String, Object> data;                    // ② 使用 Object 作为值类型 → 可以存数字、字符串、嵌套对象

    /** 消息创建时间戳（Unix 毫秒）：System.currentTimeMillis() */
    public long timestamp;

    /** 无参构造器（fastjson2 反序列化需要） */
    public MQMessage() {}

    /** 构造器：自动设置当前时间戳 */
    public MQMessage(String cmd, Map<String, Object> data) {
        this.cmd = cmd;
        this.data = data;
        this.timestamp = System.currentTimeMillis();    // ③ 自动打时间戳
    }

    /** 构造器：手动指定时间戳 */
    public MQMessage(String cmd, Map<String, Object> data, long timestamp) {
        this.cmd = cmd;
        this.data = data;
        this.timestamp = timestamp;
    }

    // ===== getter/setter =====
    public String getCmd() { return cmd; }
    public void setCmd(String cmd) { this.cmd = cmd; }
    public Map<String, Object> getData() { return data; }
    public void setData(Map<String, Object> data) { this.data = data; }
    public long getTimestamp() { return timestamp; }
    public void setTimestamp(long timestamp) { this.timestamp = timestamp; }

    /**
     * 从 data 中安全获取字符串字段
     * @param key 字段名
     * @return 字符串值，不存在则返回 null
     */
    public String getDataString(String key) {
        Object val = data != null ? data.get(key) : null;
        return val != null ? val.toString() : null;     // ④ toString() 兜底：无论什么类型都转字符串
    }

    /**
     * 从 data 中安全获取整数字段
     * @param key 字段名
     * @return 整数值，不存在则返回 null
     */
    public Integer getDataInt(String key) {
        Object val = data != null ? data.get(key) : null;
        if (val instanceof Number) return ((Number) val).intValue();  // ⑤ 已是数字类型 → 直接转换
        return val != null ? Integer.parseInt(val.toString()) : null;  // ⑥ 字符串类型 → 解析
    }

    /**
     * 从 data 中安全获取布尔字段
     * @param key 字段名
     * @return 布尔值，不存在则返回 null
     */
    public Boolean getDataBoolean(String key) {
        Object val = data != null ? data.get(key) : null;
        if (val instanceof Boolean) return (Boolean) val;
        return val != null ? Boolean.parseBoolean(val.toString()) : null;
    }

    @Override
    public String toString() {
        return "MQMessage{cmd='" + cmd + "', data=" + data + ", ts=" + timestamp + "}";
    }
}
```

## 2.1 ArgsParser.java — 命令行参数解析

```java
package inspection.common.config;

import java.util.HashMap;
import java.util.Map;

/**
 * 命令行参数解析工具
 *
 * 解析格式：--key1 value1 --key2 value2
 * 例如：--redis-host 192.168.1.100 --car-id Car001
 *
 * 用法：
 *   ArgsParser args = new ArgsParser(args);                 // 传入 main(String[] args)
 *   String host = args.get("--redis-host", "localhost");     // 获取字符串值
 *   int port = args.getInt("--redis-port", 6379);           // 获取整数值
 */
public class ArgsParser {

    /** 存储解析后的键值对：key → value */
    private final Map<String, String> map = new HashMap<>(); // ① final 保证引用不变，但 map 内容可变

    /**
     * 构造函数：解析命令行参数
     * @param args 命令行参数字符串数组
     *
     * 解析逻辑：
     *   遍历数组，遇到以 "--" 开头的字符串 → 把它当作 key
     *   key 的下一个元素 → 当作 value
     *   value 可能不存在（最后一个参数是 key）→ 忽略
     */
    public ArgsParser(String[] args) {
        for (int i = 0; i < args.length; i++) {
            // startsWith("--") 判断字符串是否以 "--" 开头
            if (args[i].startsWith("--") && i + 1 < args.length) {
                map.put(args[i], args[i + 1]);              // ② 存入 map
                i++;                                        // ③ 跳过 value（因为它不是 key）
            }
        }
    }

    /**
     * 获取字符串参数
     * @param key          参数名（含 --）
     * @param defaultValue 默认值
     * @return 参数值或默认值
     */
    public String get(String key, String defaultValue) {
        return map.getOrDefault(key, defaultValue);         // ④ JDK 8 HashMap 新方法：有则返回，无则返回默认值
    }

    /**
     * 获取整数参数
     * @param key          参数名
     * @param defaultValue 默认值
     * @return 参数值或默认值
     * @throws NumberFormatException 如果值不是有效整数
     */
    public int getInt(String key, int defaultValue) {
        String v = map.get(key);
        return v != null ? Integer.parseInt(v) : defaultValue;  // ⑤ parseInt 把字符串 "6379" 转为 int 6379
    }
}
```

## 2.3 DistributedLock.java — 分布式锁

```java
package inspection.common.client;

import redis.clients.jedis.Jedis;
import redis.clients.jedis.JedisPool;
import redis.clients.jedis.params.SetParams;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import inspection.common.config.ConfigConstants;

/**
 * 基于 Redis SET NX EX 的分布式锁
 *
 * 核心原理：
 *   利用 Redis 单线程 + SET NX EX 的原子性实现分布式锁
 *   - NX (Not eXists)：只有当 key 不存在时才 set 成功 → 互斥
 *   - EX (Expire)：设置过期时间 → 防止死锁
 *
 * 使用模式（务必在 finally 中释放锁）：
 *   DistributedLock lock = bb.getCarLock("Car001");
 *   if (lock.tryLock()) {
 *       try {
 *           // 临界区代码
 *       } finally {
 *           lock.unlock();  // ← 无论如何都要释放锁！
 *       }
 *   }
 */
public class DistributedLock {
    private static final Logger log = LoggerFactory.getLogger(DistributedLock.class);

    /** Redis 连接池（不是直接连接，每次操作借出归还） */
    private final JedisPool pool;

    /** 锁的 Redis Key */
    private final String lockKey;

    /**
     * 锁的持有者标识（线程 ID）
     * 用于安全释放：只有锁的持有者释放才有效
     * 防止线程 A 误删线程 B 的锁
     */
    private final String lockValue;

    /**
     * 构造函数
     * @param pool    Redis 连接池
     * @param lockKey 锁的 Redis Key
     */
    public DistributedLock(JedisPool pool, String lockKey) {
        this.pool = pool;
        this.lockKey = lockKey;
        this.lockValue = String.valueOf(Thread.currentThread().getId());  // ① 用线程 ID 作为锁持有者标识
    }

    /**
     * 尝试获取锁（非阻塞）
     *
     * Redis 命令：SET lockKey lockValue NX EX 30
     *   NX → 仅当 key 不存在时设置
     *   EX 30 → 30 秒后自动过期
     *
     * @return true=获取成功，false=已被其他线程/进程持有
     */
    public boolean tryLock() {
        try (Jedis jedis = pool.getResource()) {         // ② try-with-resources：自动归还连接
            SetParams params = SetParams.setParams()
                    .nx()                                 // ③ NX：只在 key 不存在时成功
                    .ex(ConfigConstants.LOCK_EXPIRE_SECONDS);  // ④ EX 30s：30 秒后自动删除
            String result = jedis.set(lockKey, lockValue, params);
            boolean locked = "OK".equals(result);         // ⑤ Redis SET NX 成功返回 "OK"，失败返回 null
            if (locked) {
                log.debug("锁获取成功: {}", lockKey);
            } else {
                log.debug("锁获取失败（已被占用）: {}", lockKey);
            }
            return locked;
        }
    }

    /**
     * 尝试获取锁（阻塞，带超时）
     *
     * 轮询重试：每隔 LOCK_RETRY_INTERVAL_MS(100ms) 尝试一次
     *
     * @param timeoutMs 最大等待毫秒
     * @return true=获取成功，false=超时
     */
    public boolean tryLock(long timeoutMs) {
        long start = System.currentTimeMillis();          // ⑥ 记录开始时间
        while (System.currentTimeMillis() - start < timeoutMs) {  // ⑦ 计算已等待时长
            if (tryLock()) return true;
            try {
                Thread.sleep(ConfigConstants.LOCK_RETRY_INTERVAL_MS);  // ⑧ 休眠 100ms 后再试
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();       // ⑨ 恢复中断标志（不要吞掉中断）
                return false;
            }
        }
        return false;                                     // ⑩ 超时，获取失败
    }

    /**
     * 释放锁（只有持有者才能释放）
     *
     * 使用 Lua 脚本保证原子性：
     *   if redis.call('get', KEYS[1]) == ARGV[1] then   ← 只有 value 匹配才删除
     *       return redis.call('del', KEYS[1])
     *   else return 0 end
     *
     * 为什么用 Lua 脚本？
     *   如果先 GET 再 DEL，这两个操作不是原子的
     *   在 GET 和 DEL 之间锁可能过期并被另一个线程获取
     *   → 线程 A 的 GET 检查通过，但 DEL 删的是线程 B 的锁 → 严重 bug！
     *   Lua 脚本在 Redis 服务端原子执行，避免此问题
     */
    public void unlock() {
        String script =
            "if redis.call('get', KEYS[1]) == ARGV[1] then " +  // ⑪ Lua：检查值是否匹配
            "return redis.call('del', KEYS[1]) " +              // ⑫ Lua：匹配则删除
            "else return 0 end";                                 // ⑬ Lua：不匹配则返回 0
        try (Jedis jedis = pool.getResource()) {
            jedis.eval(script, 1, lockKey, lockValue);          // ⑭ eval 执行 Lua 脚本
            //     参数含义：(脚本, key数量, key1, arg1, arg2, ...)
        }
        log.debug("锁已释放: {}", lockKey);
    }

    /**
     * 续期：延长锁的过期时间
     *
     * 用于需要长时间持有锁的场景
     * 同样用 Lua 脚本保证原子性
     *
     * @param seconds 续期秒数
     */
    public void renew(int seconds) {
        String script =
            "if redis.call('get', KEYS[1]) == ARGV[1] then " +
            "return redis.call('expire', KEYS[1], ARGV[2]) " + // ⑮ EXPIRE 设置过期时间
            "else return 0 end";
        try (Jedis jedis = pool.getResource()) {
            jedis.eval(script, 1, lockKey, lockValue, String.valueOf(seconds));
        }
    }

    public String getLockKey() { return lockKey; }
}
```

## 2.4 MessageBusClient.java — RabbitMQ 消息总线

```java
package inspection.common.client;

import com.alibaba.fastjson2.JSON;
import com.rabbitmq.client.*;
import inspection.common.config.ConfigConstants;
import inspection.common.model.MQMessage;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import java.io.IOException;
import java.nio.charset.StandardCharsets;
import java.util.concurrent.TimeoutException;
import java.util.function.Consumer;

/**
 * 消息总线客户端：封装全部 RabbitMQ 收发操作
 *
 * 队列拓扑（消息路由结构）：
 *   Navigator4CarID      → Navigator 共享竞争消费（多个 Navigator 轮流处理）
 *   TargetPlannerCmd     → TargetPlanner 共享竞争消费
 *   TaskConfigCmd        → TaskConfigurator 消费
 *   Car:{carId}           → 各小车独立消费
 *   UpdateView Exchange   → Fanout 广播（所有 Display 都收到）
 *
 * Controller 不订阅任何 RabbitMQ 队列（通过 Redis taskQueue 间接通信）
 */
public class MessageBusClient {
    private static final Logger log = LoggerFactory.getLogger(MessageBusClient.class);

    /** RabbitMQ 底层连接（TCP 长连接，自动恢复） */
    private Connection connection;

    /**
     * RabbitMQ Channel（轻量级虚拟连接）
     * 所有操作（publish/subscribe/declare）都通过 Channel 进行
     * Channel 不是线程安全的，所以发送操作都要加锁
     */
    private Channel channel;

    /**
     * 发送锁对象
     * RabbitMQ Channel 不是线程安全的，多个线程同时 publish 会出错
     * 所有发送操作通过 synchronized(sendLock) 保护
     */
    private final Object sendLock = new Object();

    /** 无参构造器（使用默认配置连接 localhost） */
    public MessageBusClient() {
        this(ConfigConstants.RABBITMQ_HOST, ConfigConstants.RABBITMQ_PORT,
                ConfigConstants.RABBITMQ_USER, ConfigConstants.RABBITMQ_PASS,
                ConfigConstants.RABBITMQ_VHOST);
    }

    /**
     * 全参构造器：连接 RabbitMQ
     *
     * RabbitMQ 连接是长期 TCP 连接，不需要每次操作都重新创建
     *
     * @param host   RabbitMQ 服务器地址
     * @param port   端口
     * @param user   用户名
     * @param pass   密码
     * @param vhost  虚拟主机
     */
    public MessageBusClient(String host, int port, String user, String pass, String vhost) {
        try {
            ConnectionFactory factory = new ConnectionFactory();  // ① 连接工厂
            factory.setHost(host);
            factory.setPort(port);
            factory.setUsername(user);
            factory.setPassword(pass);
            factory.setVirtualHost(vhost);
            factory.setAutomaticRecoveryEnabled(true);             // ② 自动重连：断线后自动恢复
            factory.setNetworkRecoveryInterval(5000);              // ③ 重试间隔 5 秒
            this.connection = factory.newConnection();             // ④ 建立 TCP 连接
            this.channel = connection.createChannel();             // ⑤ 创建 Channel
            log.info("MessageBusClient 已连接 RabbitMQ {}:{}", host, port);
        } catch (IOException | TimeoutException e) {
            log.error("连接 RabbitMQ 失败: {}", e.getMessage(), e);
            throw new RuntimeException("无法连接 RabbitMQ", e);    // ⑥ 连接失败直接抛运行时异常（程序无法继续）
        }
    }

    /** 获取底层 Channel（供 TaskConfigurator 手动确认消息等场景使用） */
    public Channel getChannel() {
        return channel;
    }

    // ==================== 队列与 Exchange 声明 ====================

    /**
     * 声明一个持久化队列
     *
     * @param queueName 队列名
     * @throws IOException RabbitMQ 通信异常
     */
    public void declareQueue(String queueName) throws IOException {
        synchronized (sendLock) {                                 // ⑦ 加锁保护 Channel 线程安全
            channel.queueDeclare(
                queueName,  // 队列名
                true,       // durable=true：RabbitMQ 重启后队列不丢失
                false,      // exclusive=false：允许多个消费者共享
                false,      // autoDelete=false：没有消费者也不删除
                null        // 无额外参数
            );
        }
        log.info("队列已声明: {}", queueName);
    }

    /**
     * 声明 Fanout Exchange
     *
     * Fanout 类型：不关心 routing key，消息广播给所有绑定的队列
     *
     * @param exchangeName Exchange 名称
     */
    public void declareFanoutExchange(String exchangeName) throws IOException {
        synchronized (sendLock) {
            channel.exchangeDeclare(
                exchangeName,
                BuiltinExchangeType.FANOUT,  // ⑧ FANOUT = 广播模式
                true                         // durable=true：持久化
            );
        }
        log.info("Fanout Exchange 已声明: {}", exchangeName);
    }

    /** 绑定队列到 Fanout Exchange */
    public void bindQueueToExchange(String queueName, String exchangeName) throws IOException {
        synchronized (sendLock) {
            channel.queueBind(queueName, exchangeName, "");      // ⑨ routingKey=""（Fanout 忽略它）
        }
    }

    /**
     * 声明所有系统队列和 Exchange（TaskConfigurator 初始化时调用一次）
     *
     * 这是整个系统的"通信基础设施初始化"，必须在其他模块启动前完成
     */
    public void declareAllSystemQueues() throws IOException {
        synchronized (sendLock) {
            declareQueue(ConfigConstants.QUEUE_NAVIGATOR_4_CAR_ID);    // Navigator 命令队列
            declareQueue(ConfigConstants.QUEUE_TARGET_PLANNER_CMD);    // TargetPlanner 命令队列
            declareQueue(ConfigConstants.QUEUE_TASK_CONFIG_CMD);       // TaskConfigurator 命令队列
            for (int i = 1; i <= 16; i++) {                             // 预声明 16 个小车队列
                declareQueue(ConfigConstants.carQueueName(
                    String.format("Car%03d", i)));                      // ⑩ Car001~Car016
            }
            declareFanoutExchange(ConfigConstants.EXCHANGE_UPDATE_VIEW); // Display 刷新广播
        }
        log.info("所有系统队列/Exchange 已声明");
    }

    // ==================== 消息发送 ====================

    /**
     * 发送 MQMessage 到指定队列
     *
     * @param queueName 目标队列名
     * @param message   MQMessage 对象
     */
    public void sendToQueue(String queueName, MQMessage message) {
        try {
            String json = JSON.toJSONString(message);              // ⑪ fastjson2 序列化为 JSON
            synchronized (sendLock) {
                channel.basicPublish(
                    "",                               // ⑫ exchange="" → 使用默认 Exchange（直连队列）
                    queueName,                        // ⑬ routingKey=队列名
                    MessageProperties.PERSISTENT_TEXT_PLAIN,  // ⑭ 持久化消息（RabbitMQ 重启不丢）
                    json.getBytes(StandardCharsets.UTF_8)     // ⑮ JSON 字符串 → UTF-8 字节数组
                );
            }
            log.debug("发送 → {}: {}", queueName, message.getCmd());
        } catch (IOException e) {
            log.error("发送消息失败 → {}: {}", queueName, e.getMessage(), e);
        }
    }

    /**
     * 发送原始 JSON 字符串（兼容队友代码的 publish 方法名）
     */
    public void publish(String queueName, String message) {
        sendRaw(queueName, message);
    }

    /**
     * 发送原始 JSON 字符串
     *
     * 兜底声明队列：防止 RabbitMQ 重启后队列丢失导致消息被静默丢弃
     */
    public void sendRaw(String queueName, String json) {
        try {
            synchronized (sendLock) {
                channel.queueDeclare(queueName, true, false, false, null);  // ⑯ 兜底声明（幂等操作）
                channel.basicPublish("", queueName,
                        MessageProperties.PERSISTENT_TEXT_PLAIN,
                        json.getBytes(StandardCharsets.UTF_8));
            }
            log.debug("发送原始 → {}", queueName);
        } catch (IOException e) {
            log.error("发送原始消息失败 → {}: {}", queueName, e.getMessage(), e);
        }
    }

    /**
     * 广播原始字符串到 Fanout Exchange
     *
     * 所有绑定到此 Exchange 的队列都会收到消息
     */
    public void fanoutPublish(String exchangeName, String message) {
        try {
            synchronized (sendLock) {
                channel.basicPublish(
                    exchangeName,  // ⑰ 此时第一个参数是 Exchange 名而非 ""
                    "",            // ⑱ routingKey=""（Fanout 忽略 routing key）
                    MessageProperties.PERSISTENT_TEXT_PLAIN,
                    message.getBytes(StandardCharsets.UTF_8)
                );
            }
            log.debug("Fanout广播 → {}", exchangeName);
        } catch (IOException e) {
            log.error("Fanout广播失败 → {}: {}", exchangeName, e.getMessage(), e);
        }
    }

    /** 广播 MQMessage 到 Fanout Exchange */
    public void broadcast(String exchangeName, MQMessage message) {
        try {
            String json = JSON.toJSONString(message);
            synchronized (sendLock) {
                channel.basicPublish(exchangeName, "",
                        MessageProperties.PERSISTENT_TEXT_PLAIN,
                        json.getBytes(StandardCharsets.UTF_8));
            }
        } catch (IOException e) {
            log.error("广播失败 → {}: {}", exchangeName, e.getMessage(), e);
        }
    }

    // ==================== 消息订阅 ====================

    /**
     * 订阅指定队列（自动确认 + 原始字符串回调）
     */
    public void subscribeText(String queueName, Consumer<String> handler) {
        try {
            declareQueue(queueName);
            channel.basicQos(1);  // ⑲ 公平分发：每次只预取 1 条消息
            DeliverCallback deliverCallback = (consumerTag, delivery) -> {
                String body = new String(delivery.getBody(), StandardCharsets.UTF_8);
                try {
                    handler.accept(body);  // ⑳ 调用用户提供的处理函数
                } catch (Exception e) {
                    log.error("处理消息异常: {}", e.getMessage(), e);
                }
            };
            channel.basicConsume(queueName, true, deliverCallback, consumerTag -> {});
            // ⑳+1   autoAck=true → 自动确认（消息发出即认为已处理）
        } catch (IOException e) {
            log.error("订阅队列失败", e);
        }
    }

    /**
     * 订阅指定队列（自动确认 + MQMessage 对象回调）
     */
    public void subscribe(String queueName, Consumer<MQMessage> handler) {
        subscribe(queueName, handler, true);
    }

    /**
     * 订阅指定队列
     *
     * @param queueName 队列名
     * @param handler   消息处理回调
     * @param autoAck   true=自动确认，false=手动确认
     */
    public void subscribe(String queueName, Consumer<MQMessage> handler, boolean autoAck) {
        try {
            declareQueue(queueName);
            channel.basicQos(1);  // 公平分发
            DeliverCallback deliverCallback = (consumerTag, delivery) -> {
                String body = new String(delivery.getBody(), StandardCharsets.UTF_8);
                try {
                    MQMessage msg = JSON.parseObject(body, MQMessage.class);  // JSON 反序列化
                    if (msg != null) {
                        handler.accept(msg);
                    }
                } catch (Exception e) {
                    log.error("处理消息异常: {}", e.getMessage(), e);
                } finally {
                    if (!autoAck) {  // 手动确认模式
                        channel.basicAck(
                            delivery.getEnvelope().getDeliveryTag(),  // 消息的唯一 delivery tag
                            false  // false=只确认这一条，不批量确认
                        );
                    }
                }
            };
            channel.basicConsume(queueName, autoAck, deliverCallback, consumerTag -> {});
        } catch (IOException e) {
            log.error("订阅队列失败", e);
        }
    }

    /**
     * 订阅 Fanout Exchange 广播消息
     *
     * 创建一个临时的、独占的、自动删除的队列，绑定到 Exchange
     * 当这个客户端断开连接时，临时队列自动删除
     *
     * @param exchangeName Exchange 名称
     * @param handler      消息处理回调
     * @return 临时队列名称（debug 用）
     */
    public String subscribeFanout(String exchangeName, Consumer<MQMessage> handler) {
        try {
            declareFanoutExchange(exchangeName);
            String queueName = channel.queueDeclare().getQueue();  // 创建临时队列（随机名称）
            channel.queueBind(queueName, exchangeName, "");         // 绑定到 Fanout Exchange
            channel.basicConsume(queueName, true, (consumerTag, delivery) -> {
                String body = new String(delivery.getBody(), StandardCharsets.UTF_8);
                try {
                    MQMessage msg = JSON.parseObject(body, MQMessage.class);
                    if (msg != null) handler.accept(msg);
                } catch (Exception e) {
                    log.error("广播消息处理异常", e);
                }
            }, consumerTag -> {});
            return queueName;
        } catch (IOException e) {
            log.error("订阅 Exchange 失败", e);
            return null;
        }
    }

    // ==================== 管理 ====================

    /** 关闭连接（先关 Channel 再关 Connection） */
    public void close() {
        try {
            if (channel != null && channel.isOpen()) channel.close();
            if (connection != null && connection.isOpen()) connection.close();
            log.info("MessageBusClient 已关闭");
        } catch (IOException | TimeoutException e) {
            log.error("关闭连接异常", e);
        }
    }
}
```

```java
 * │ pause:operator:{id}  │ String    │ 个人暂停标志                 │
 * │ replay:snapshots     │ List      │ 回放快照列表                 │
 * └─────────────────────┴──────────┴──────────────────────────────┘
 */
public class BlackboardClient {
    private static final Logger log = LoggerFactory.getLogger(BlackboardClient.class);

    /** Redis 连接池（线程安全，重用 TCP 连接） */
    private final JedisPool pool;

    /** 本地缓存的地图宽度（从 config:task 读取，最多每秒刷新一次） */
    private int mapWidth;

    /** 本地缓存的地图高度 */
    private int mapHeight;

    /** 上次从 Redis 刷新地图尺寸的时间戳（防频繁读取） */
    private long lastConfigRefresh = 0;

    // ============= Bitmap 本地缓存 =============
    // 由于 bitmap 读取频繁（每 100ms 一次），但写入不频繁（只在车移动时写入）
    // 使用版本号缓存避免重复从 Redis GET 整个 bitmap

    /** 缓存的 map:view bitmap 原始字节 */
    private byte[] cachedMapViewBytes;

    /** 缓存时的 map:view bitmap 版本号（与 Redis 对比判断是否失效） */
    private long cachedMapViewVersion = -1;

    /** 缓存的 map:blocked bitmap 原始字节 */
    private byte[] cachedMapBlockedBytes;

    /** 缓存时的 map:blocked bitmap 版本号 */
    private long cachedMapBlockedVersion = -1;

    /** 强制失效所有 bitmap 缓存（跨进程感知，其他进程写 bitmap 后调用） */
    public void invalidateBitmapCache() {
        this.cachedMapViewBytes = null;
        this.cachedMapViewVersion = -1;
        this.cachedMapBlockedBytes = null;
        this.cachedMapBlockedVersion = -1;
    }

    // ==================== 构造方法 ====================

    /** 默认尺寸构造器 */
    public BlackboardClient(String host, int port) {
        this(host, port, ConfigConstants.DEFAULT_MAP_WIDTH, ConfigConstants.DEFAULT_MAP_HEIGHT);
    }

    /**
     * 全参构造器：创建 Redis 连接池
     *
     * 连接池参数说明：
     *   maxTotal=20     → 最多 20 个并发连接（我们的场景够用）
     *   maxIdle=5       → 池中最多闲置 5 个连接
     *   minIdle=1       → 至少保持 1 个空闲连接（"热身"）
     *   testOnBorrow    → 借出前测试连通性（安全但略有性能开销）
     *   testWhileIdle   → 后台定期检测空闲连接是否存活
     */
    public BlackboardClient(String host, int port, int mapWidth, int mapHeight) {
        JedisPoolConfig config = new JedisPoolConfig();
        config.setMaxTotal(20);
        config.setMaxIdle(5);
        config.setMinIdle(1);
        config.setTestOnBorrow(true);
        config.setTestWhileIdle(true);
        config.setTimeBetweenEvictionRuns(Duration.ofSeconds(30));
        config.setMinEvictableIdleDuration(Duration.ofSeconds(60));
        this.pool = new JedisPool(config, host, port,
                ConfigConstants.REDIS_TIMEOUT_MS, null);  // null = 无密码
        this.mapWidth = mapWidth;
        this.mapHeight = mapHeight;
        log.info("BlackboardClient 已连接 Redis {}:{}", host, port);
    }

    // ==================== 地图配置刷新 ====================

    /**
     * 从 Redis config:task 刷新地图尺寸（最多每秒一次）
     *
     * 频率限制的原因：
     *   此方法在 getMapWidth/getMapHeight/getExploredPercent 等高频方法中调用
     *   但地图尺寸很少变化，每秒读一次 Redis 就够了
     *   避免在高频 tick 中对 Redis 产生不必要的 HGET 压力
     */
    public void refreshMapConfig() {
        long now = System.currentTimeMillis();
        if (now - lastConfigRefresh < 1000) return;          // ① 不到 1 秒，跳过
        try (Jedis jedis = pool.getResource()) {
            String w = jedis.hget(ConfigConstants.KEY_TASK_CONFIG, "mapWidth");
            String h = jedis.hget(ConfigConstants.KEY_TASK_CONFIG, "mapHeight");
            if (w != null) mapWidth = Integer.parseInt(w);
            if (h != null) mapHeight = Integer.parseInt(h);
        } catch (Exception e) {
            // config:task 可能还不存在（初始化前），忽略异常
        }
        lastConfigRefresh = now;
    }

    /**
     * 手动设置地图尺寸（TaskConfigurator 初始化时调用）
     * 跳过了 refreshMapConfig 的频率限制
     */
    public void setMapSize(int w, int h) {
        this.mapWidth = w;
        this.mapHeight = h;
        this.lastConfigRefresh = System.currentTimeMillis();  // ② 更新时间戳，防止短时间内再次刷新
    }

    /** 获取底层 Jedis 连接（DistributedLock、bgsave 等场景使用） */
    public Jedis getJedis() {
        return pool.getResource();
    }

    public int getMapWidth()  { refreshMapConfig(); return mapWidth; }
    public int getMapHeight() { refreshMapConfig(); return mapHeight; }

    /** 读取 bitmap 版本号 */
    private long getBitmapVersion(String versionKey) {
        try (Jedis jedis = pool.getResource()) {
            String val = jedis.get(versionKey);
            return val != null ? Long.parseLong(val) : 0;
        }
    }

    /** 递增 bitmap 版本号 → 通知所有缓存失效 */
    private void incrementBitmapVersion(String versionKey) {
        try (Jedis jedis = pool.getResource()) {
            jedis.incr(versionKey);  // ③ INCR 原子递增，不必担心并发
        }
    }

    // ==================== Bitmap 编解码 ====================

    /**
     * 坐标 → bitmap 偏移（行优先编码）
     *
     * 示例：40×40 地图，(x=2, y=1) → offset = 1*40+2 = 42
     * Redis SETBIT 的 offset 就是这个值
     *
     * @return long 类型（Redis BIT 操作接受 long 偏移）
     */
    private long bitmapOffset(int x, int y) {
        return (long) y * mapWidth + x;
    }

    /**
     * 从 Redis 读取 bitmap 原始字节
     * 处理了 null 和长度不足的情况
     *
     * 需要的字节数 = (width × height + 7) / 8
     *   40×40=1600 bits → (1600+7)/8 = 200 字节
     */
    private byte[] getBitmapBytes(String key) {
        try (Jedis jedis = pool.getResource()) {
            byte[] data = jedis.get(key.getBytes());          // ④ 以二进制方式 GET（不是文本）
            int expectedBytes = (mapWidth * mapHeight + 7) / 8;
            if (data == null) {
                data = new byte[expectedBytes];               // ⑤ Redis 中不存在 → 返回全 0 数组
            } else if (data.length < expectedBytes) {
                byte[] newData = new byte[expectedBytes];
                System.arraycopy(data, 0, newData, 0, data.length);  // ⑥ 补齐不足的字节
                data = newData;
            }
            return data;
        } catch (Exception e) {
            log.warn("getBitmapBytes failed for {}", key, e);
            return new byte[(mapWidth * mapHeight + 7) / 8];
        }
    }

    /**
     * 字节数组解码为 boolean 二维数组
     *
     * Redis bitmap 的位序：
     *   offset 0 → 第 0 字节的 bit 7（最高位 MSB）
     *   offset 1 → 第 0 字节的 bit 6
     *   ...
     *   offset 7 → 第 0 字节的 bit 0（最低位 LSB）
     *   offset 8 → 第 1 字节的 bit 7
     *
     * 解码方法：(data[byteIdx] & (1 << (7 - offset%8))) != 0
     */
    private boolean[][] decodeBitmap(byte[] data, int width, int height) {
        if (data == null) return new boolean[height][width];
        boolean[][] grid = new boolean[height][width];
        for (int y = 0; y < height; y++) {
            for (int x = 0; x < width; x++) {
                int offset = y * width + x;
                int byteIdx = offset / 8;                    // ⑦ 第几个字节
                if (byteIdx < data.length) {
                    grid[y][x] = (data[byteIdx] & (1 << (7 - (offset % 8)))) != 0;
                    // ⑧ (1 << n) 是位运算：1 左移 n 位，生成只有第 n 位为 1 的掩码
                    //    & 是按位与：如果那一位是 1 → 结果非 0 → true
                }
            }
        }
        return grid;
    }

    // ==================== 地图探索 ====================

    /**
     * 获取探索状态二维数组（带版本号缓存）
     *
     * 缓存逻辑：
     *   1. 读 Redis version key
     *   2. 如果本地缓存版本 != Redis 版本 → 缓存失效 → 重新 GET + 重新解码
     *   3. 如果版本相同 → 直接返回缓存的解码结果
     *
     * 带来的性能提升：
     *   避免每次 tick 都做：网络 GET（~0.5ms）+ 1600 次位解码（~0.5ms）
     *   缓存命中后：直接返回 boolean[][]（~0.001ms）
     */
    public boolean[][] getMapView() {
        refreshMapConfig();
        long currentVer = getBitmapVersion(ConfigConstants.KEY_MAP_VIEW_VERSION);
        if (cachedMapViewBytes == null || cachedMapViewVersion != currentVer) {
            cachedMapViewBytes = getBitmapBytes(ConfigConstants.KEY_MAP_VIEW);
            cachedMapViewVersion = currentVer;
        }
        return decodeBitmap(cachedMapViewBytes, mapWidth, mapHeight);
    }

    /** 设置单个格子的探索状态 */
    public void setMapViewBit(int x, int y) {
        try (Jedis jedis = pool.getResource()) {
            jedis.setbit(ConfigConstants.KEY_MAP_VIEW, bitmapOffset(x, y), true);
            removeFromUnexploredSet(jedis, x, y);             // ⑨ 同步移除未探索索引
            incrementBitmapVersion(ConfigConstants.KEY_MAP_VIEW_VERSION);
        }
        cachedMapViewBytes = null;                            // ⑩ 使本地缓存失效
        cachedMapViewVersion = -1;
    }

    /** GETBIT 单个检查 */
    public boolean isExplored(int x, int y) {
        try (Jedis jedis = pool.getResource()) {
            return jedis.getbit(ConfigConstants.KEY_MAP_VIEW, bitmapOffset(x, y));
        }
    }

    /**
     * 3×3 区域点亮（Pipeline 批量 + 原子性）
     *
     * 关键设计：跳过障碍物格
     *   obstacle 永远不可探索，不应该被标为"已探索"
     */
    public void illuminateArea(int cx, int cy) {
        refreshMapConfig();
        int r = ConfigConstants.ILLUMINATE_RADIUS;
        try (Jedis jedis = pool.getResource()) {
            var pipeline = jedis.pipelined();                 // ⑪ 开启 Pipeline（批量命令模式）
            for (int dy = -r; dy <= r; dy++) {
                for (int dx = -r; dx <= r; dx++) {
                    int nx = cx + dx;
                    int ny = cy + dy;
                    if (nx >= 0 && nx < mapWidth
                            && ny >= 0 && ny < mapHeight
                            && !isBlocked(nx, ny)) {          // ⑫ 跳过障碍物
                        pipeline.setbit(ConfigConstants.KEY_MAP_VIEW,
                                bitmapOffset(nx, ny), true);
                        pipeline.srem(ConfigConstants.KEY_UNEXPLORED_SET,
                                nx + "," + ny);
                    }
                }
            }
            pipeline.incr(ConfigConstants.KEY_MAP_VIEW_VERSION);
            pipeline.sync();                                  // ⑬ 一次性发送所有命令 → 1 次网络往返
        }
        cachedMapViewBytes = null;
        cachedMapViewVersion = -1;
    }

    // ==================== 障碍物 ====================

    public boolean isBlocked(int x, int y) {
        refreshMapConfig();
        try (Jedis jedis = pool.getResource()) {
            return jedis.getbit(ConfigConstants.KEY_MAP_BLOCKED, bitmapOffset(x, y));
        }
    }

    /**
     * Pipeline 批量写入障碍物（TaskConfigurator 专用）
     *
     * 160 个障碍物 × (SETBIT + SREM) = 320 条 Redis 命令
     * 不用 Pipeline：320 × 0.5ms = 160ms
     * 用了 Pipeline：1 × 0.5ms = 0.5ms
     * 性能提升：~320x
     */
    public void setBlockedBatch(List<Point> obstacles) {
        refreshMapConfig();
        try (Jedis jedis = pool.getResource()) {
            var pipeline = jedis.pipelined();
            for (Point obs : obstacles) {
                pipeline.setbit(ConfigConstants.KEY_MAP_BLOCKED,
                        bitmapOffset(obs.x, obs.y), true);
                pipeline.srem(ConfigConstants.KEY_UNEXPLORED_SET,
                        obs.x + "," + obs.y);                // ⑭ 障碍物格从"未探索"集合移除
            }
            pipeline.incr(ConfigConstants.KEY_MAP_BLOCKED_VERSION);
            pipeline.sync();
        }
        cachedMapBlockedBytes = null;
        cachedMapBlockedVersion = -1;
    }

    public void setBlocked(int x, int y) {
        refreshMapConfig();
        try (Jedis jedis = pool.getResource()) {
            jedis.setbit(ConfigConstants.KEY_MAP_BLOCKED, bitmapOffset(x, y), true);
            removeFromUnexploredSet(jedis, x, y);
            incrementBitmapVersion(ConfigConstants.KEY_MAP_BLOCKED_VERSION);
        }
        cachedMapBlockedBytes = null;
        cachedMapBlockedVersion = -1;
    }

    /** 获取所有障碍物坐标（遍历 bitmap 解码） */
    public List<Point> getAllBlocked() {

        return list;
    }

    // ==================== 路径操作（核心：List + LPUSH/RPOP） ====================

    /**
     * 推入整条路径（Navigator 调用）
     *
     * 存储方式：LPUSH 从队首推入 → List 中路径是从终点到起点的逆序
     * 消费方式：RPOP 从队尾弹出一个点 → 每次弹的最后 push 的，即从起点开始走
     *
     * 示例：路径 [(10,10)→(11,10)→(12,10)→(13,10)]
     *   LPUSH: (10,10)(11,10)(12,10)(13,10) → List: [13,10],[12,10],[11,10],[10,10]
     *   RPOP: 弹出 [10,10] → 第一步从 (10,10) 开始
     */
    public void pushRoute(String carId, List<Point> path) {
        try (Jedis jedis = pool.getResource()) {
            String key = ConfigConstants.carRouteKey(carId);
            jedis.del(key);                                   // ⑱ 先清理旧路径
            if (path != null && !path.isEmpty()) {
                String[] jsons = new String[path.size()];
                for (int i = 0; i < path.size(); i++) {
                    jsons[i] = JSON.toJSONString(path.get(i));
                }
                jedis.lpush(key, jsons);                       // ⑲ 一次性 LPUSH 整个数组
            }
        }
    }

    /** 弹下一步（Car 调用）：RPOP → 消费路径的最后一个元素 */
    public Point popNextStep(String carId) {
        try (Jedis jedis = pool.getResource()) {
            String json = jedis.rpop(ConfigConstants.carRouteKey(carId));
            return json != null ? JSON.parseObject(json, Point.class) : null;
        }
    }

    /**
     * 查看下一步但不消费（LRANGE -1 -1）
     * Controller 和 Car 都用这个判断"有没有路要走"
     * 不修改路径本身
     */
    public Point peekNextStep(String carId) {
        try (Jedis jedis = pool.getResource()) {
            List<String> list = jedis.lrange(ConfigConstants.carRouteKey(carId), -1, -1);
            if (list.isEmpty()) return null;
            return JSON.parseObject(list.get(0), Point.class);
        }
    }

    /** 获取完整路径 */
    public List<Point> getFullRoute(String carId) {
        List<Point> path = new ArrayList<>();
        try (Jedis jedis = pool.getResource()) {
            List<String> jsons = jedis.lrange(ConfigConstants.carRouteKey(carId), 0, -1);
            for (int i = jsons.size() - 1; i >= 0; i--) {   // ⑳ 逆序遍历（因为 List 是 LPUSH 的逆序）
                path.add(JSON.parseObject(jsons.get(i), Point.class));
            }
        }
        return path;
    }

    // ==================== 探索统计 ====================

    /**
     * 已探索格子数
     * BITCOUNT map:view → O(N) 在 Redis 中（N=字节数）

    public int getExploredCount() {

    /**
     * 探索百分比（排除障碍物格）
     *
     * 公式：exploredCount / (totalCells - obstacleCount) × 100
     *
     * 这个公式是正确的：障碍物永远不可探索，所以分母不应包含它们
     * 例如：1600 格，160 个障碍物 → 分母=1440
     *       如果探索了 720 格 → 探索率 = 720/1440 = 50%
     */
    public double getExploredPercent() {
        refreshMapConfig();
        int obstacleCount = getObstacleCount();
        int explorable = mapWidth * mapHeight - obstacleCount;
        if (explorable <= 0) return 100.0;
        double pct = getExploredCount() * 100.0 / explorable;
        return Math.min(100.0, pct);  // 兜底：防止浮点误差导致超过 100%
    }

    /**

    /**

     *
     * 与全量初始化的区别：
     *   全量：clearAll() → 重新生成一切
     *   增量：只创建新车数据，不动地图和已有小车

    public void addCar(String carId, int x, int y) {
        try (Jedis jedis = pool.getResource()) {

            jedis.hset(ConfigConstants.carPositionKey(carId), "x", String.valueOf(x));

            illuminateArea(x, y);
            pushTask("ROUTE_NEEDED", carId, null);
            log.info("已增量添加小车: carId={}, position=({},{})", carId, x, y);
        }
    }

    /**
     * 增量移除小车
     * 1. 清除该车当前位置的障碍物标记（车本身就是"移动障碍物"）
     * 2. 删除所有 car:{id}:* 键
     * 3. 从 cars 列表中移除
     */
    public void removeCar(String carId) {
        try (Jedis jedis = pool.getResource()) {
            // 清理动态障碍物标记
            Point pos = getCarPosition(carId);
            if (pos != null) {
                jedis.setbit(ConfigConstants.KEY_MAP_BLOCKED,
                        bitmapOffset(pos.x, pos.y), false);   // ⑳+1 清除该车占的位
                jedis.incr(ConfigConstants.KEY_MAP_BLOCKED_VERSION);
            }
            // 删除所有相关 key
            jedis.del(ConfigConstants.carStatusKey(carId));
            jedis.del(ConfigConstants.carPositionKey(carId));
            jedis.del(ConfigConstants.carStepsKey(carId));
            jedis.del(ConfigConstants.carTargetKey(carId));
            jedis.del(ConfigConstants.carRouteKey(carId));
            jedis.del(ConfigConstants.carBlockedTickKey(carId));
            jedis.del(ConfigConstants.carLockKey(carId));
            // 更新 cars 列表
            String carsJson = jedis.hget(ConfigConstants.KEY_TASK_CONFIG, "cars");
            if (carsJson != null && !carsJson.isEmpty()) {
                List<String> cars = new ArrayList<>(JSON.parseArray(carsJson, String.class));
                cars.remove(carId);
                jedis.hset(ConfigConstants.KEY_TASK_CONFIG, "cars", JSON.toJSONString(cars));
            }
            log.info("已移除小车: carId={}", carId);
        }
        cachedMapBlockedBytes = null;
        cachedMapBlockedVersion = -1;
    }

    /**
     * 向 taskQueue 队尾 RPUSH 一个任务（带 carId）
     *
     * RPUSH → 从右侧（尾部）插入
     * LPOP  → 从左侧（头部）弹出
     * 形成先进先出（FIFO）队列
     */
    public void pushTask(String taskType, String carId, Map<String, String> extra) {
        try (Jedis jedis = pool.getResource()) {
            Map<String, String> task = new LinkedHashMap<>();  // LinkedHashMap 保持插入顺序（JSON 友好）
            task.put("type", taskType);
            task.put("carId", carId);
            if (extra != null) task.putAll(extra);
            jedis.rpush(ConfigConstants.KEY_TASK_QUEUE, JSON.toJSONString(task));
        }
    }

    /** 推送全局任务（无 carId） */
    public void pushTask(String taskType, Map<String, String> extra) {
        try (Jedis jedis = pool.getResource()) {
            Map<String, String> task = new LinkedHashMap<>();
            task.put("type", taskType);
            if (extra != null) task.putAll(extra);
            jedis.rpush(ConfigConstants.KEY_TASK_QUEUE, JSON.toJSONString(task));
        }
    }

    /**
     * 从 taskQueue 队首 LPOP 一个任务（非阻塞）
     * @return 任务 Map，队列为空时返回 null
     */
    @SuppressWarnings("unchecked")                           // 压制 fastjson2 泛型警告
    public Map<String, String> popTask() {
        try (Jedis jedis = pool.getResource()) {
            String json = jedis.lpop(ConfigConstants.KEY_TASK_QUEUE);
            if (json == null) return null;
            return JSON.parseObject(json, Map.class);
        }
    }

    /**
     * BLPOP 阻塞式出队（事件驱动核心）
     *
     * 与 LPOP 的区别：
     *   LPOP：队列空了立即返回 null（轮询模式）
     *   BLPOP：队列空了阻塞等待，直到有新元素或超时（事件驱动）
     *
     * @param timeoutSeconds 超时秒数
     * @return 任务 Map，超时返回 null
     */
    @SuppressWarnings("unchecked")
    public Map<String, String> blockingPopTask(int timeoutSeconds) {
        try (Jedis jedis = pool.getResource()) {
            List<String> result = jedis.blpop(timeoutSeconds, ConfigConstants.KEY_TASK_QUEUE);
            // BLPOP 返回 [key, value] 的 List
            // 注意不是 [key, value]，而是包含这两个元素的 list
            // result.get(0) == "taskQueue", result.get(1) == JSON
            if (result == null || result.size() < 2) return null;
            return JSON.parseObject(result.get(1), Map.class);
        }
    }

    /**
     * 初始化未探索 Set（TaskConfigurator 专用）
     * 将全部 (x,y) 坐标以 "x,y" 格式 SADD 到 unexplored:set
     */
    public void initUnexploredSet(int width, int height) {
        try (Jedis jedis = pool.getResource()) {
            String key = ConfigConstants.KEY_UNEXPLORED_SET;
            jedis.del(key);
            String[] members = new String[width * height];
            int idx = 0;
            for (int y = 0; y < height; y++)
                for (int x = 0; x < width; x++)
                    members[idx++] = x + "," + y;
            // 分批 SADD（每批 1000 个）
            int batchSize = 1000;
            for (int i = 0; i < members.length; i += batchSize) {
                int end = Math.min(i + batchSize, members.length);
                jedis.sadd(key, java.util.Arrays.copyOfRange(members, i, end));
            }
        }
    }
```

## 3.1 TaskConfiguratorMain.java — 逐行完整注解

```java
package inspection.taskconfigurator;                     // ① 包：按照功能模块组织

// ===== import 区：引入需要使用的其他类 =====
import com.alibaba.fastjson2.JSON;                       // ② fastjson2 JSON 工具（序列化/反序列化）
import com.alibaba.fastjson2.JSONObject;                  //    JSONObject：可读写的 JSON 键值对
import com.rabbitmq.client.Channel;                       // ③ RabbitMQ Channel（队列操作）
import com.rabbitmq.client.DeliverCallback;               //    消息到达时的回调接口
import inspection.common.client.BlackboardClient;         // ④ 黑板客户端（Redis）
import inspection.common.client.MessageBusClient;         // ⑤ 消息总线客户端（RabbitMQ）
import inspection.common.config.ArgsParser;               // ⑥ 命令行参数解析
import inspection.common.config.ConfigConstants;          // ⑦ 全局常量
import inspection.common.enums.CarStatus;                 // ⑧ 小车状态枚举
import inspection.common.model.MQMessage;                 // ⑨ 消息模型
import inspection.common.model.Point;                    // ⑩ 坐标模型
import org.slf4j.Logger;                                 // ⑪ 日志接口
import org.slf4j.LoggerFactory;                           // ⑫ 日志工厂
import java.io.IOException;                              // ⑬ IO 异常
import java.nio.charset.StandardCharsets;                 // ⑭ 标准字符集（UTF-8）
import java.util.*;                                      // ⑮ List, Map, Set, Random 等

/**
 * 任务配置器：负责初始化整个仿真环境
 *
 * 生命周期：
 *   1. 启动 → 监听 RabbitMQ TaskConfigCmd 队列
 *   2. 收到 FORWARD_CONFIG 消息 → 增量操作或全量初始化
 *   3. 永久运行（事件驱动，不主动退出）
 *
 * 调用方：Controller（通过 RabbitMQ），Display（通过 Redis taskQueue → Controller 转发）
 */
public class TaskConfiguratorMain {

    /** SLF4J Logger：日志输出（替代 System.out.println） */
    private static final Logger LOG = LoggerFactory.getLogger(TaskConfiguratorMain.class);

    /** 监听的 RabbitMQ 队列名（与 ConfigConstants.QUEUE_TASK_CONFIG_CMD 值相同） */
    private static final String TASK_CONFIG_QUEUE = "TaskConfigCmd";

    /** Redis 黑板客户端（读写的"粉笔"） */
    private BlackboardClient blackboard;

    /** RabbitMQ 消息总线（收发的"对讲机"） */
    private MessageBusClient messageBus;

    /**
     * 伪随机数生成器
     * final → 只创建一次，复用以提高随机性和性能
     * Random 在单线程下是线程安全的（我们的场景中只用这一个线程）
     */
    private final Random random = new Random();

    // ======================== 入口 ========================

    /**
     * main 方法：Java 程序的标准入口
     *
     * 只有一行代码：创建对象 → 调用启动
     * 这种"入口归入口，逻辑归逻辑"的模式让代码干净清晰
     *
     * @param args 命令行参数
     * @throws Exception 启动失败时向上抛（程序崩溃退出）
     */
    public static void main(String[] args) throws Exception {
        new TaskConfiguratorMain().start(args);
        // ⑯ new TaskConfiguratorMain() → 调用无参构造器创建对象
        // ⑰ .start(args)               → 立即调用启动方法
    }

    // ======================== 启动方法 ========================

    /**
     * 启动 TaskConfigurator
     *
     * 执行顺序：
     *   1. 解析命令行参数（获取 Redis/RabbitMQ 地址）
     *   2. 创建 BlackboardClient（连接 Redis）
     *   3. 创建 MessageBusClient（连接 RabbitMQ）
     *   4. 声明自己的命令队列
     *   5. 设置消息监听回调（注册 handleForwardConfig）
     *   6. 开始消费消息（阻塞在 RabbitMQ 事件循环）
     *
     * @param args 命令行参数数组
     * @throws Exception 连接失败等异常
     */
    public void start(String[] args) throws Exception {

        // ——— 1. 解析命令行参数 ———
        ArgsParser argsParser = new ArgsParser(args);                       // ⑱ 创建解析器
        String redisHost = argsParser.get("--redis-host",                  // ⑲ 获取参数
                ConfigConstants.REDIS_HOST);                                //    默认值 "localhost"
        int redisPort = argsParser.getInt("--redis-port",
                ConfigConstants.REDIS_PORT);                                // ⑳ 默认 6379
        String mqHost = argsParser.get("--mq-host",
                ConfigConstants.RABBITMQ_HOST);
        int mqPort = argsParser.getInt("--mq-port",
                ConfigConstants.RABBITMQ_PORT);

        // ——— 2. 创建客户端 ———
        // BlackboardClient 构造函数内部创建 JedisPool（Redis 连接池）
        blackboard = new BlackboardClient(redisHost, redisPort);
        // MessageBusClient 构造函数内部创建 RabbitMQ Connection + Channel
        messageBus = new MessageBusClient(mqHost, mqPort,
                ConfigConstants.RABBITMQ_USER, ConfigConstants.RABBITMQ_PASS,
                ConfigConstants.RABBITMQ_VHOST);

        // ——— 3. 声明队列 ———
        Channel channel = messageBus.getChannel();                          // ⑳+1 获取 RabbitMQ Channel
        channel.queueDeclare(                                               // ⑳+2 声明队列（幂等操作）
                TASK_CONFIG_QUEUE,  // 队列名："TaskConfigCmd"
                true,               // durable：持久化（RabbitMQ 重启不丢）
                false,              // exclusive：非独占
                false,              // autoDelete：不自动删除
                null                // 无额外参数
        );

        LOG.info("任务配置器已启动，监听队列: {}", TASK_CONFIG_QUEUE);       // ⑳+3 {} 是 SLF4J 占位符

        // ——— 4. 设置消息处理回调 ———
        // DeliverCallback 是 RabbitMQ 的回调接口：
        //   当 TaskConfigCmd 队列有新消息时，RabbitMQ 自动调用此回调
        DeliverCallback deliverCallback = (consumerTag, delivery) -> {
            // ⑳+4 Lambda 表达式：(参数) -> { 函数体 }
            // consumerTag：RabbitMQ 分配的消费者标签（这里没用）
            // delivery：消息内容对象

            // ⑳+5 把字节数组转为 UTF-8 字符串
            String message = new String(delivery.getBody(), StandardCharsets.UTF_8);
            try {
                handleForwardConfig(message);  // ⑳+6 核心处理逻辑
            } catch (Exception e) {
                LOG.error("处理消息失败: {}", message, e);
            } finally {
                // ⑳+7 手动确认消息已处理（autoAck=false）
                // delivery.getEnvelope().getDeliveryTag() → 消息的唯一 ID
                // false → 只确认这一条（不批量确认）
                channel.basicAck(delivery.getEnvelope().getDeliveryTag(), false);
            }
        };

        // ——— 5. 开始消费 ———
        // basicConsume 参数：
        //   TASK_CONFIG_QUEUE  → 订阅的队列名
        //   false              → autoAck=false（手动确认）
        //   deliverCallback    → 收到消息时回调
        //   consumerTag -> {}  → 消费者注册成功后的回调（这里为空）
        channel.basicConsume(TASK_CONFIG_QUEUE, false, deliverCallback,
                consumerTag -> {});
        // ⑳+8 这个方法调用后，RabbitMQ 开始向此 Channel 推送消息
        //     当前线程会阻塞在 RabbitMQ 的内部事件循环中
        //     程序不会退出，一直等待新消息
    }

    /**
     * 通过反射读取 ConfigConstants 中的字段值（备用于参数补全）
     *
     * Java 反射：在运行时动态读取类的字段和方法
     * 这里用于读取 ConfigConstants 中的静态常量
     *
     * @param fieldName    字段名（如 "DEFAULT_MAP_WIDTH"）
     * @param defaultValue 如果反射失败返回的默认值
     * @return 常量值的字符串形式
     */
    private String getConfigOrDefault(String fieldName, String defaultValue) {
        try {
            java.lang.reflect.Field field = ConfigConstants.class.getField(fieldName);
            // ⑳+9 getField() 获取 public 字段
            //          get(null) 获取静态字段的值（静态字段不属于任何实例，所以传 null）
            Object value = field.get(null);
            return value != null ? value.toString() : defaultValue;
        } catch (Exception e) {
            return defaultValue;  // ⑳+10 反射失败（比如字段不存在）→ 返回默认值
        }
    }

    // ======================== 核心处理方法 ========================

    /**
     * 处理 FORWARD_CONFIG 消息（最核心的方法，约 120 行）
     *
     * 消息来源：Controller 通过 RabbitMQ Queue TaskConfigCmd 发送
     *
     * 处理分三种情况：
     *   A. 增量添加小车（addCar=true）         → 只新增一辆车，不重建地图
     *   B. 增量移除小车（removeCar=true）      → 只移除一辆车，不重建地图
     *   C. 全量初始化（forceReset=true 或首次） → clearAll + 重建一切
     *
     * @param messageJson RabbitMQ 收到的原始 JSON 字符串
     */
    private void handleForwardConfig(String messageJson) {

        // ——— 解析消息 ———
        // fastjson2 反序列化：JSON 字符串 → MQMessage 对象
        MQMessage msg = JSON.parseObject(messageJson, MQMessage.class);

        // 获取命令类型
        String cmd = msg.getCmd();

        // 防御性检查：只处理 FORWARD_CONFIG 命令
        // 注意写法是 "FORWARD_CONFIG".equals(cmd) 而非 cmd.equals("FORWARD_CONFIG")
        // 原因：如果 cmd==null，前者返回 false（安全），后者抛 NullPointerException
        if (!"FORWARD_CONFIG".equals(cmd)) {
            LOG.warn("忽略非 FORWARD_CONFIG 命令: {}", cmd);
            return;  // ⑳+11 直接返回，不做任何处理
        }

        // getData() 返回 Map<String, Object> → 转 JSON 字符串 → 再转 JSONObject
        // JSONObject 提供了 getIntValue/getBooleanValue 等便利方法
        JSONObject data = JSONObject.parseObject(JSON.toJSONString(msg.getData()));

        // ======== 分支 A：增量添加小车 ========
        boolean addCar = data.getBooleanValue("addCar", false);
        // ⑳+12 getBooleanValue(key, default)：从 JSON 取布尔值，不存在则用默认值
        if (addCar) {
            String carId = data.getString("addCarId");         // 要添加的小车 ID
            int cx = data.getIntValue("x", ConfigConstants.DEFAULT_MAP_WIDTH / 2);  // 出生点 X
            int cy = data.getIntValue("y", ConfigConstants.DEFAULT_MAP_HEIGHT / 2);  // 出生点 Y
            if (carId != null && !carId.isEmpty()) {
                blackboard.addCar(carId, cx, cy);              // ⑳+13 在 Redis 中创建小车全部数据
                // 记录车归属（多机协作时哪个运行员添加的这辆车）
                String machine = data.getString("machine");
                if (machine != null && !machine.isEmpty()) {
                    blackboard.setCarOwner(carId, machine);
                }
                // 为新车声明 RabbitMQ 队列（如 Car:Car005）
                try {
                    messageBus.getChannel().queueDeclare(
                            ConfigConstants.carQueueName(carId), true, false, false, null);
                } catch (IOException e) { LOG.error("声明 {} 队列失败", carId, e); }
                LOG.info("增量添加小车完成: carId={}, position=({},{})", carId, cx, cy);
            }
            return;  // ⑳+14 增量操作到此结束！不再执行后续的全量初始化代码
        }

        // ======== 分支 B：增量移除小车 ========
        boolean removeCar = data.getBooleanValue("removeCar", false);
        if (removeCar) {
            String carId = data.getString("removeCarId");
            if (carId != null && !carId.isEmpty()) {
                blackboard.removeCar(carId);                   // 删除 Redis 中该车的所有数据
                LOG.info("增量移除小车完成: carId={}", carId);
            }
            return;  // ⑳+15 增量操作到此结束
        }

        // ======== 分支 C：全量初始化（含崩溃恢复判断）========

        // 检查是否强制重置
        boolean forceReset = data.getBooleanValue("forceReset", false);

        // 读取 Redis 中是否已有配置
        Map<String, String> existingConfig = blackboard.getTaskConfig();

        // 崩溃恢复逻辑：
        //   如果 !forceReset（非强制重置）且已有配置 → 说明上次初始化后 TaskConfigurator 崩溃重启
        //   → 跳过全量初始化，复用 Redis 中已有的地图和小车
        if (!forceReset && existingConfig != null && !existingConfig.isEmpty()) {
            LOG.info("检测到已有配置 (forceReset=false)，跳过全量初始化，复用现有状态");
            try {
                blackboard.getJedis().bgsave();  // ⑳+16 BGSAVE 持久化当前 Redis 快照
            } catch (Exception e) {
                // 持久化失败不影响主流程（/* ignore */）
            }
            return;
        }

        // ——— 读取参数（从消息中提取配置值） ———
        int mapWidth = data.getIntValue("mapWidth",
                ConfigConstants.DEFAULT_MAP_WIDTH);           // ⑳+17 地图宽，默认 40
        int mapHeight = data.getIntValue("mapHeight",
                ConfigConstants.DEFAULT_MAP_HEIGHT);           //    地图高，默认 40
        int carCount = data.getIntValue("carCount", 1);       //    小车数，默认 1
        Double od = data.getDouble("obstacleDensity");        // ⑳+18 障碍物密度
        double obstacleDensity = od != null ? od
                : ConfigConstants.DEFAULT_OBSTACLE_DENSITY;   //    不存在则用默认 0.1(10%)

        LOG.info("收到配置: mapWidth={}, mapHeight={}, carCount={}, density={}",
                mapWidth, mapHeight, carCount, obstacleDensity);

        // ──── 全量初始化流程（6 步）────

        // 第 1 步：清空 Redis（删除所有项目相关 key）
        blackboard.clearAll();
        blackboard.setMapSize(mapWidth, mapHeight);            // 设置本地地图尺寸缓存
        LOG.info("Redis 已清空，地图尺寸: {}x{}", mapWidth, mapHeight);

        // 第 2 步：初始化未探索集合（全部 (x,y) → unexplored:set）
        blackboard.initUnexploredSet(mapWidth, mapHeight);

        // 第 3 步：计算网格分区出生点
        List<Point> spawnPoints = computeSpawnPoints(mapWidth, mapHeight, carCount);

        // 第 4 步：构建禁区（出生点 3×3 区域不生成障碍物）
        Set<Point> forbidden = buildForbiddenZone(spawnPoints, mapWidth, mapHeight);

        // 第 5 步：生成随机障碍物
        int obstacleCount = generateObstacles(mapWidth, mapHeight, obstacleDensity, forbidden);

        // 第 6 步：初始化小车
        List<String> carIds = new ArrayList<>();               // ⑳+19 存储所有小车的 ID 列表
        for (int i = 1; i <= carCount; i++) {
            String carId = String.format("Car%03d", i);        // ⑳+20 格式化为 "Car001", "Car002", ...
            carIds.add(carId);                                 // 加入 ID 列表
            Point spawn = spawnPoints.get(i - 1);              // ⑳+21 取出对应的出生点
            blackboard.setCarPosition(carId, spawn.x, spawn.y);// 写入位置
            blackboard.setCarStatus(carId, CarStatus.IDLE);    // 写入初始状态（空闲）
            blackboard.setCarSteps(carId, 0);                  // 写入初始步数（0）
            blackboard.illuminateArea(spawn.x, spawn.y);       // 点亮出生点及周围 3×3
            LOG.info("已初始化小车: carId={}, position=({},{})",
                    carId, spawn.x, spawn.y);
        }

        // 第 7 步：写入任务配置到 Redis config:task Hash
        Map<String, String> config = new HashMap<>();
        config.put("mapWidth", String.valueOf(mapWidth));      // ⑳+22 valueOf 将 int 转 String
        config.put("mapHeight", String.valueOf(mapHeight));
        config.put("carCount", String.valueOf(carCount));
        config.put("cars", JSON.toJSONString(carIds));         // ⑳+23 List → JSON数组 → String
        config.put("obstacleDensity", String.valueOf(obstacleDensity));
        String routeAlgorithm = data.getString("routeAlgorithm");
        config.put("routeAlgorithm", routeAlgorithm != null ? routeAlgorithm : "BFS");
        boolean startActive = data.getBooleanValue("active", false);
        config.put("taskActive", startActive ? "1" : "0");    // ⑳+24 初始化后是否自动激活
        blackboard.setTaskConfig(config);                      // ⑳+25 写入 Redis

        // 第 7.5 步：初始化 FIFO 任务队列
        blackboard.clearTaskQueue();                           // ⑳+26 先清空旧任务
        for (String carId : carIds) {
            blackboard.pushTask("ROUTE_NEEDED", carId, null);  // ⑳+27 每车推一个"需要规划路径"
        }
        LOG.info("已初始化 taskQueue: {} 个 ROUTE_NEEDED 任务", carIds.size());

        // 第 8 步：声明所有 RabbitMQ 队列和 Exchange
        try {
            messageBus.declareAllSystemQueues();               // ⑳+28 预声明 16 个小车队列 + Navigator 等
        } catch (IOException e) {
            LOG.error("声明系统队列失败: {}", e.getMessage(), e);
        }

        LOG.info("初始化完成: mapWidth={}, mapHeight={}, carCount={}, obstacleCount={}, "
                + "taskQueue 已填入 {} 个 ROUTE_NEEDED 任务。等待用户 Start 激活 Controller",
                mapWidth, mapHeight, carCount, obstacleCount, carIds.size());

        // 第 9 步：持久化 Redis 快照（用于崩溃恢复）
        try {
            blackboard.getJedis().bgsave();                    // ⑳+29 后台 fork 子进程写 RDB 文件
        } catch (Exception e) {
            // 持久化失败不影响主流程
        }
    }

    // ======================== 辅助算法方法 ========================

    /**
     * 计算网格分区出生点
     *
     * 目标：将 N 辆车尽可能均匀地分散到 W×H 地图的不同区域
     *
     * 算法步骤：
     *   1. cols = ⌈√N⌉（列数取平方根上取整，尽量正方形）
     *   2. rows = ⌈N/cols⌉（行数补足）
     *   3. cellW = W/cols, cellH = H/rows（每格尺寸）
     *   4. 每辆车放在对应格子的中心点，并限制在 [1, W-2] × [1, H-2] 边距内
     *
     * 数学示例：N=4, W=40, H=40
     *   cols = ⌈2⌉ = 2, rows = ⌈4/2⌉ = 2
     *   cellW = 20, cellH = 20
     *   Car001 → (10, 10)
     *   Car002 → (30, 10)
     *   Car003 → (10, 30)
     *   Car004 → (30, 30)
     *
     * @param mapWidth  地图宽
     * @param mapHeight 地图高
     * @param carCount  小车数量
     * @return 出生点列表（长度 = carCount）
     */
    private List<Point> computeSpawnPoints(int mapWidth, int mapHeight, int carCount) {
        if (carCount <= 0) return new ArrayList<>();           // ⑳+30 防御：0 或负数返回空列表

        int cols = (int) Math.ceil(Math.sqrt(carCount));      // ⑳+31 列数 = ⌈√N⌉
        int rows = (int) Math.ceil((double) carCount / cols); // ⑳+32 行数 = ⌈N/cols⌉
        int cellW = mapWidth / cols;                           // ⑳+33 每格宽（整数除法，向下取整）
        int cellH = mapHeight / rows;                          //    每格高

        LOG.info("网格分区: {}x{} 地图, {} 辆车 → {}列×{}行, 每格 {}×{}",
                mapWidth, mapHeight, carCount, cols, rows, cellW, cellH);

        List<Point> spawns = new ArrayList<>();
        int idx = 0;
        // 双重循环遍历网格
        for (int row = 0; row < rows && idx < carCount; row++) {      // ⑳+34 条件 includes idx<carCount
            for (int col = 0; col < cols && idx < carCount; col++) {  //     确保不超过车数边界
                int cx = col * cellW + cellW / 2;   // ⑳+35 格子中心 X = 左边界 + 半宽
                int cy = row * cellH + cellH / 2;   //        格子中心 Y = 上边界 + 半高

                // 边界修剪：确保出生点离地图边缘至少 1 格
                cx = Math.max(1, Math.min(mapWidth - 2, cx));        // ⑳+36 限制在 [1, W-2]
                cy = Math.max(1, Math.min(mapHeight - 2, cy));       //        限制在 [1, H-2]

                spawns.add(new Point(cx, cy));
                idx++;
            }
        }
        return spawns;
    }

    /**
     * 构建禁区集合：所有出生点及其 3×3 周围不得生成障碍物
     *
     * 因为如果出生点旁边有障碍物，车一出生就被困住了
     *
     * @param spawnPoints 出生点列表
     * @param mapWidth    地图宽
     * @param mapHeight   地图高
     * @return 禁区坐标集合（HashSet）
     */
    private Set<Point> buildForbiddenZone(List<Point> spawnPoints,
                                           int mapWidth, int mapHeight) {
        Set<Point> forbidden = new HashSet<>();                    // ⑳+37 HashSet 提供 O(1) 的 contains 查询
        for (Point spawn : spawnPoints) {
            for (int dy = -1; dy <= 1; dy++) {                    // ⑳+38 遍历 3×3：±1 偏移
                for (int dx = -1; dx <= 1; dx++) {
                    int nx = spawn.x + dx;
                    int ny = spawn.y + dy;
                    // 确保在地图范围内
                    if (nx >= 0 && nx < mapWidth && ny >= 0 && ny < mapHeight) {
                        forbidden.add(new Point(nx, ny));
                    }
                }
            }
        }
        LOG.info("禁区集合: {} 个出生点, {} 个禁止放置格",
                spawnPoints.size(), forbidden.size());
        return forbidden;
    }

    /**
     * 随机生成障碍物（避开禁区）
     *
     * 采用"拒绝采样"算法：
     *   1. 随机选一个坐标
     *   2. 检查是否在禁区或已放置
     *   3. 不在 → 接受；在 → 拒绝，重新随机
     *   4. 重复直到放够目标数量或达到最大尝试次数
     *
     * @param mapWidth        地图宽
     * @param mapHeight       地图高
     * @param obstacleDensity 障碍物密度（如 0.1 = 10%）
     * @param forbidden       禁区集合
     * @return 实际放置的障碍物数量
     */
    private int generateObstacles(int mapWidth, int mapHeight,
                                   double obstacleDensity, Set<Point> forbidden) {
        int totalCells = mapWidth * mapHeight;                     // ⑳+39 总格子数：40×40=1600
        int targetObstacleCount = (int) (totalCells * obstacleDensity);  // 目标数：1600×0.1=160

        int placed = 0;                                            // 已放置计数
        int maxAttempts = targetObstacleCount * 20;               // ⑳+40 最大尝试次数 = 3200
        int attempts = 0;                                          // 已尝试次数
        java.util.List<Point> obstacles = new java.util.ArrayList<>();  // 临时收集障碍物坐标

        while (placed < targetObstacleCount && attempts < maxAttempts) {
            int x = random.nextInt(mapWidth);                      // ⑳+41 生成 [0, W-1] 的随机整数
            int y = random.nextInt(mapHeight);
            Point p = new Point(x, y);
            attempts++;

            // 检查条件：不在禁区 且 该位置还没有障碍物
            if (!forbidden.contains(p) && !blackboard.isBlocked(x, y)) {
                obstacles.add(p);
                placed++;
            }
        }

        if (placed < targetObstacleCount) {
            LOG.warn("障碍物生成不足: 目标={}, 实际={}", targetObstacleCount, placed);
        }

        // Pipeline 批量写入障碍物到 Redis
        blackboard.setBlockedBatch(obstacles);                     // ⑳+42 一次性写入（非逐条）
        LOG.info("已生成 {} 个障碍物 (目标: {})", placed, targetObstacleCount);
        return placed;
    }
}
```

```java
     *

     *   1. 解析命令行参数
     *   2. 创建 BlackboardClient（连 Redis）
     *   3. 创建 MessageBusClient（连 RabbitMQ）
     *   4. 创建并启动 ControllerAgent
     *   5. 注册关闭钩子（Ctrl+C 时优雅退出）

            //    instanceId    → 从 0 开始的本实例编号
            //    totalInstances → 总共运行多少个 Controller 实例
            int instanceId = argsParser.getInt("--instance-id", 0);
            int totalInstances = argsParser.getInt("--total-instances", 1);

            // ——— 2. 创建客户端 ———
            BlackboardClient blackboard = new BlackboardClient(redisHost, redisPort);
            MessageBusClient messageBus = new MessageBusClient(mqHost, mqPort,
                    ConfigConstants.RABBITMQ_USER, ConfigConstants.RABBITMQ_PASS,
                    ConfigConstants.RABBITMQ_VHOST);

            // ④ instanceId+1 → 让用户看到从 1 开始的编号（人类友好）

            Runtime.getRuntime().addShutdownHook(new Thread(() -> {

                        instanceId + 1, totalInstances);
                controller.stop();           // ⑦ 先停止 ControllerAgent（停止线程）
                try {
                    messageBus.close();      // ⑧ 关闭 RabbitMQ 连接

                    log.error("关闭 MessageBus 失败", e);
                }
                blackboard.close();          // ⑨ 关闭 Redis 连接池
                log.info("Controller {}/{} stopped.",
                        instanceId + 1, totalInstances);
            }));

        } catch (Exception e) {
            log.error("Controller 启动失败", e);
            System.exit(1);                  // ⑩ 启动失败 → 异常退出码 1（脚本可检测）
        }
    }
```

## 3.3 ControllerAgent.java — 核心调度器逐行注解

```java
package inspection.controller;

import com.alibaba.fastjson2.JSONObject;                    // ① fastjson2 JSONObject
import inspection.common.client.BlackboardClient;           // ② 黑板客户端（Redis）
import inspection.common.client.DistributedLock;            // ③ 分布式锁
import inspection.common.client.MessageBusClient;           // ④ 消息总线（RabbitMQ）
import inspection.common.config.ConfigConstants;            // ⑤ 全局常量
import inspection.common.enums.CarStatus;                   // ⑥ 小车状态枚举
import inspection.common.enums.CommandType;                 // ⑦ 命令类型枚举
import inspection.common.model.Point;                      // ⑧ 坐标模型
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

import java.util.*;                                        // ⑨ List, Map, Arrays 等
import java.util.concurrent.Executors;                     // ⑩ 线程池工厂
import java.util.concurrent.ScheduledExecutorService;      // ⑪ 定时任务调度器
import java.util.concurrent.TimeUnit;                      // ⑫ 时间单位（MILLISECONDS 等）

/**
 * Controller 核心调度器（C2 架构的"大脑"）
 *
 * 设计原则：
 *   1. 不监听任何 RabbitMQ 队列（区别于其他模块）
 *   2. 通过 Redis taskQueue 拉取任务（BLPOP 阻塞模式）
 *   3. 根据任务类型通过 RabbitMQ 分发命令
 *   4. 每 100ms 执行一次周期广播（推进车、刷新前端）
 *
 * 线程模型：
 *   - taskProcessor 线程：BLPOP 等待 + 处理任务
 *   - broadcastScheduler 线程：每 100ms 执行广播
 *   - 这两个线程通过 volatile 变量 taskActive 协调
 */
public class ControllerAgent {
    private static final Logger log = LoggerFactory.getLogger(ControllerAgent.class);

    // ==================== 核心依赖 ====================

    /** Redis 黑板客户端（读数据、写状态） */
    private final BlackboardClient bb;

    /** RabbitMQ 消息总线（发命令给 Navigator、Car、Display） */
    private final MessageBusClient mq;

    // ==================== 线程基础设施 ====================

    /**
     * 广播调度器：单线程定时执行
     * scheduleAtFixedRate(this::broadcastTick, 100ms, 100ms)
     * 如果 broadcastTick 执行时间超过 100ms，下次执行会延迟（但我们的 tick 只需要 ~5ms）
     */
    private final ScheduledExecutorService broadcastScheduler =
            Executors.newSingleThreadScheduledExecutor();

    /**
     * 任务唤醒锁对象（Java 并发原语）
     *
     * 用于 taskProcessLoop 中的 wait/notify 协调：
     *   - taskProcessLoop 在 taskActive=false 时调用 wait(1000) 休眠
     *   - handleStartTask 和 stop 通过 notifyAll() 立即唤醒
     *
     * 为什么用一个专门的 Object 而不是用 this？
     *   使用单独的锁对象更精确，避免外部代码 unknownly 进入 synchronized(this) 造成意外阻塞
     */
    private final Object taskWakeLock = new Object();

    // ==================== 多实例分片 ====================

    /** 当前实例的编号（0-based） */
    private final int instanceId;

    /** 总实例数 */
    private final int totalInstances;

    // ==================== 运行状态（volatile 保证多线程可见性） ====================

    /**
     * volatile → 保证变量的读写对所有线程立即可见
     * 如果不用 volatile，一个线程修改了 running=false，
     * 另一个线程可能因为 CPU 缓存看到的是旧的 true 值
     */
    private volatile boolean running = false;

    /** 任务是否激活（true=正在处理任务，false=等待激活） */
    private volatile boolean taskActive = false;

    /**
     * 用户是否主动激活了任务
     * 区分于 taskActive：
     *   taskActive=false 可能是用户点了"暂停"或"探索自动完成"
     *   userActivated 区分这两种情况
     */
    private volatile boolean userActivated = false;

    /** 任务处理线程的引用（用于 stop 时 join 等待） */
    private Thread taskProcessor;

    /** 节拍计数器（只在 broadcastScheduler 线程中递增，不需要 volatile） */
    private long tickCount = 0;

    /** 是否在录制快照（volatile → 可能被 processTask 中的 RECORD_START/STOP 修改） */
    private volatile boolean recording = false;

    // ==================== 构造函数 ====================

    /** 默认单实例构造器 */
    public ControllerAgent(BlackboardClient bb, MessageBusClient mq) {
        this(bb, mq, 0, 1);                                 // ⑬ 链式调用另一个构造器
    }

    /** 全参构造器 */
    public ControllerAgent(BlackboardClient bb, MessageBusClient mq,
                           int instanceId, int totalInstances) {
        this.bb = bb;
        this.mq = mq;
        this.instanceId = instanceId;
        this.totalInstances = totalInstances;
    }

    // ==================== 启动 / 停止 ====================

    /**
     * 启动 ControllerAgent
     *
     * 做两件事：
     *   1. 启动 taskProcessor 线程（事件驱动任务处理）
     *   2. 启动 broadcastScheduler（周期 100ms）
     */
    public void start() {
        running = true;

        // 清除上次运行可能残留的状态（防止崩溃重启后的脏数据）
        bb.setTaskActive(false);                             // ⑭ 写入 Redis：taskActive=false
        bb.clearTaskQueue();                                 // ⑮ 清空旧任务队列

        // 启动任务处理线程
        taskProcessor = new Thread(this::taskProcessLoop,    // ⑯ 方法引用：指向 taskProcessLoop 方法
                "controller-task-processor");                  //   线程名称（便于调试）
        taskProcessor.setDaemon(true);                       // ⑰ 守护线程 → JVM 退出时不等待此线程
        taskProcessor.start();                               // ⑱ 启动线程（调用 taskProcessLoop）

        // 启动 100ms 定时广播
        broadcastScheduler.scheduleAtFixedRate(
                this::broadcastTick,                         // ⑲ 方法引用：指向 broadcastTick 方法
                ConfigConstants.TICK_INTERVAL_MS,             //   首次延迟 100ms
                ConfigConstants.TICK_INTERVAL_MS,             //   间隔 100ms
                TimeUnit.MILLISECONDS                         //   时间单位为毫秒
        );

        log.info("Controller 已启动: 纯 Redis taskQueue 驱动 + {}ms 周期性广播",
                ConfigConstants.TICK_INTERVAL_MS);
    }

    /**
     * 停止 ControllerAgent
     *
     * 关闭顺序：
     *   1. 设置 running=false（taskProcessLoop 会退出循环）
     *   2. 设置 taskActive=false
     *   3. 唤醒 taskProcessLoop（如果它在 wait 中）
     *   4. 停止广播调度器（不再执行新 tick）
     *   5. 等待 taskProcessor 线程结束（最多 2 秒）
     */
    public void stop() {
        running = false;
        taskActive = false;
        wakeTaskProcessor();                                 // ⑳ 唤醒可能在 wait 中的 taskProcessLoop
        broadcastScheduler.shutdown();                       // ⑳+1 停止定时器
        if (taskProcessor != null) {
            try {
                taskProcessor.join(2000);                    // ⑳+2 等待线程结束，最多等 2 秒
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();          // ⑳+3 恢复中断标志
            }
        }
    }

    // ==================== 任务处理主循环（最核心的方法） ====================

    /**
     * 任务处理主循环：Controller 的"心跳"
     *
     * 运行模型：
     *   while (running) {
     *     if (!taskActive) {
     *       wait(1000)；          ← 没激活就睡
     *       检查 Redis taskActive；← 醒来看看是不是被激活了
     *       continue;
     *     }
     *     BLPOP 2秒阻塞等任务；   ← 有激活就拉任务
     *     processTask；           ← 处理
     *     连续 LPOP 清空剩余任务；← 防积压
     *   }
     */
    private void taskProcessLoop() {
        log.info("任务处理线程启动 (纯 Redis BLPOP), taskActive={}", taskActive);

        while (running) {                                    // ⑳+4 只要没被 stop 就继续循环
            try {

                // ===== 阶段 1：等待激活 =====
                if (!taskActive) {
                    synchronized (taskWakeLock) {             // ⑳+5 获取 taskWakeLock 的监视器锁
                        // 双重检查：防止在获取锁之前 taskActive 被改为 true
                        if (!taskActive && running) {
                            taskWakeLock.wait(1000);          // ⑳+6 释放锁，休眠最多 1 秒
                            // wait() 被以下情况唤醒：
                            //   a) wakeTaskProcessor() 调用 notifyAll()
                            //   b) 超时 1 秒
                        }
                    }
                    // ⑳+7 离开 synchronized 块，自动释放锁

                    // 超时唤醒后检查 Redis：Display 可能直接写 Redis 激活了任务
                    if (!taskActive && running) {
                        try {
                            boolean redisActive = bb.isTaskActive();
                            if (redisActive) {
                                log.info("检测到 Redis taskActive=true, 激活任务处理器");
                                taskActive = true;           // ⑳+8 更新本地状态
                            }
                        } catch (Exception e) {
                            // Redis 可能暂时不可用，下个循环重试
                        }
                    }
                    continue;                                // ⑳+9 回到 while(running) 循环开头
                }

                // ===== 阶段 2：拉取并处理任务 =====
                // BLPOP 阻塞等待任务（队列为空时最多等 2 秒）
                Map<String, String> task = bb.blockingPopTask(2);

                if (task != null) {
                    processTask(task);                       // ⑳+10 分发处理

                    // 连续清空剩余任务（避免积压在队列中）
                    // 因为 BLPOP 只能拉一条，但期间可能有新任务被 RPUSH 进来
                    Map<String, String> next;
                    while ((next = bb.popTask()) != null) {  // ⑳+11 非阻塞 LPOP 直到队列空
                        processTask(next);
                    }
                }

            } catch (InterruptedException e) {
                // 被 stop() 或 shutdown 中断 → 退出循环
                Thread.currentThread().interrupt();          // ⑳+12 恢复中断标志
                break;
            } catch (Exception e) {
                log.error("任务处理异常", e);
                try {
                    Thread.sleep(500);                       // ⑳+13 冷却 500ms 后重试
                } catch (InterruptedException ie) {
                    Thread.currentThread().interrupt();
                    break;
                }
            }
        }
        log.info("任务处理线程退出");
    }

    /**
     * 唤醒任务处理线程（从 wait 中立即唤醒）
     *
     * 调用时机：
     *   - handleStartTask：用户点击"开始"
     *   - broadcastTick：探索完成
     *   - stop：程序关闭
     */
    private void wakeTaskProcessor() {
        synchronized (taskWakeLock) {                        // ⑳+14 必须持有锁才能 notifyAll
            taskWakeLock.notifyAll();                         // ⑳+15 唤醒所有等待的线程
        }
    }

    // ==================== 任务分发 ====================

    /**
     * 任务分发中心：根据任务类型路由到对应的处理方法
     *
     * 分两类：
     *   全局任务（无 carId）：START, PAUSE, SET_CONFIG, RESET, RECORD_START/STOP
     *   车辆任务（有 carId）：ROUTE_NEEDED, MOVE_READY, ADD_CAR, REMOVE_CAR, BLOCKED
     *
     * @param task 从 Redis taskQueue 取出的任务 Map {type, carId, ...}
     */
    private void processTask(Map<String, String> task) {
        String type = task.get("type");
        String carId = task.get("carId");

        // ===== 全局任务（不需要 carId）=====
        switch (type) {
            case "START":
                handleStartTask();                           // → 激活任务 / 触发初始化
                return;
            case "PAUSE":
                handlePauseTask();                           // → 停止任务处理
                return;
            case "SET_CONFIG":
                handleSetConfigTask(task);                   // → 转发配置（不重建）
                return;
            case "RESET":
                handleResetTask(task);                       // → 转发重置（重建）
                return;
            case "RECORD_START":
                log.info("开始录制快照");
                recording = true;                            // ⑳+16 volatile 写，所有线程可见
                return;
            case "RECORD_STOP":
                log.info("停止录制快照");
                recording = false;
                return;
        }

        // ===== 车辆任务 =====
        if (carId == null) {
            log.warn("taskQueue 任务缺少 carId: {}", task);
            return;                                          // ⑳+17 格式错误的任务直接丢弃
        }

        // 多实例分片：只处理分配给自己的车
        if (!isMyCar(getAllCarIds(), carId)) return;         // ⑳+18 不归我管 → 跳过

        switch (type) {
            case "ROUTE_NEEDED":
                // 需要路径规划 → 发 NAVIGATE 命令给 Navigator
                log.info("[ROUTE_NEEDED] 发 NAVIGATE → Navigator4CarID, carId={}", carId);
                requestNavigate(carId);
                break;

            case "MOVE_READY":
                // 路径就绪 → 发 MOVE_STEP 命令给小车
                // ——— 双层暂停检查 ———
                if (bb.isGlobalPaused()) {                   // ⑳+19 第 1 层：全局暂停
                    log.debug("全局暂停中，跳过 carId={}", carId);
                    break;
                }
                String owner = bb.getCarOwner(carId);
                if (owner != null && !owner.isEmpty()
                        && bb.isOperatorPaused(owner)) {      // ⑳+20 第 2 层：个人暂停
                    log.debug("运行员 {} 暂停中，跳过 carId={}", owner, carId);
                    break;
                }
                // 通过暂停检查 → 发移动命令
                log.info("[MOVE_READY] 发 MOVE_STEP → Car:{}", carId);
                JSONObject moveData = new JSONObject();
                moveData.put("carId", carId);
                sendCommand(CommandType.MOVE_STEP, moveData,
                        ConfigConstants.carQueueName(carId));  // ⑳+21 发给指定小车队列
                break;

            case "ADD_CAR":
                // 新增小车 → 触发导航
                log.info("[ADD_CAR] 新增小车: carId={}，触发导航", carId);
                requestNavigate(carId);
                break;

            case "REMOVE_CAR":
                log.info("[REMOVE_CAR] 移除小车: carId={}，已从系统中注销", carId);
                break;

            case "BLOCKED":
                handleBlockedTimeout(carId);                 // ⑳+22 受阻超时恢复
                break;

            default:
                log.warn("taskQueue 未知任务类型: {}", type);
        }
    }

    // ==================== 全局任务处理 ====================

    /**
     * 处理 START 任务：用户点了"开始探索"
     *
     * 两个分支：
     *   A. 配置不存在（用户直接点"开始"，没先配地图）
     *      → 自动生成默认配置 → 发 FORWARD_CONFIG(active=true) 给 TaskConfigurator
     *      → TaskConfigurator 初始化完成后直接激活任务（taskActive=true）
     *
     *   B. 配置已存在（用户先点了"应用配置"再点"开始"）
     *      → 直接激活任务 → wakeTaskProcessor 立即开始处理
     */
    private void handleStartTask() {
        Map<String, String> existingConfig = bb.getTaskConfig();
        boolean hasConfig = existingConfig != null && !existingConfig.isEmpty();

        log.info("Start: hasConfig={}, redisTaskActive={}, 当前taskActive={}",
                hasConfig, bb.isTaskActive(), taskActive);

        if (!hasConfig) {
            // ===== 分支 A：从未初始化 → 触发全量初始化 =====
            log.warn("配置不存在，触发完整初始化");
            userActivated = true;
            JSONObject initData = new JSONObject();
            initData.put("mapWidth", bb.getMapWidth());      // 默认 40
            initData.put("mapHeight", bb.getMapHeight());    // 默认 40
            initData.put("carCount", Math.max(4, bb.getAllCarIds().size()));  // 至少 4 辆
            initData.put("obstacleDensity", ConfigConstants.DEFAULT_OBSTACLE_DENSITY);
            initData.put("routeAlgorithm", bb.getRouteAlgorithm());  // 默认 BFS
            initData.put("active", true);                     // ← 初始化后直接激活！
            taskActive = false;                                // 先暂停自己，等 TaskConfigurator 完成
            sendCommand(CommandType.FORWARD_CONFIG, initData,
                    ConfigConstants.QUEUE_TASK_CONFIG_CMD);
        } else {
            // ===== 分支 B：配置已存在 → 直接激活 =====
            userActivated = true;
            bb.setTaskActive(true);                            // ⑳+23 写入 Redis
            taskActive = true;                                 // ⑳+24 更新本地（立即生效）
            wakeTaskProcessor();                               // ⑳+25 唤醒任务处理线程
        }
        // 自动开始录制快照
        recording = true;
        log.info("自动开始录制快照");
    }

    /**
     * 处理 PAUSE 任务：用户点了"暂停"
     *
     * 注意：全局暂停和操作员暂停的底层处理在 Display/CommandReceiver 中
     * （通过 bb.setGlobalPause 或 bb.setOperatorPause 写 Redis）
     * Controller 只需要停止本地 taskActive 来停止任务处理
     */
    private void handlePauseTask() {
        log.info("Pause: 停用任务处理器");
        userActivated = false;
        taskActive = false;                                  // ⑳+26 停止任务处理
        // 暂停时保持录制（暂停状态也是回放的一部分）
    }

    /**
     * 处理 SET_CONFIG 任务：用户点了"应用配置"
     *
     * 与 RESET 的区别：不强制重建（forceReset=false）
     * TaskConfigurator 端检测到已有配置 → 跳过初始化
     *
     * ← 注意：这意味着 SET_CONFIG 实际上不会改变地图参数
     *   当前实现中参数变化需要配合 RESET
     */
    private void handleSetConfigTask(Map<String, String> task) {
        JSONObject data = new JSONObject();
        data.put("mapWidth", Integer.parseInt(
                task.getOrDefault("mapWidth",
                        String.valueOf(ConfigConstants.DEFAULT_MAP_WIDTH))));
        data.put("mapHeight", Integer.parseInt(
                task.getOrDefault("mapHeight",
                        String.valueOf(ConfigConstants.DEFAULT_MAP_HEIGHT))));
        data.put("carCount", Integer.parseInt(
                task.getOrDefault("carCount",
                        String.valueOf(Math.max(4, bb.getAllCarIds().size())))));
        data.put("obstacleDensity", Double.parseDouble(
                task.getOrDefault("obstacleDensity",
                        String.valueOf(ConfigConstants.DEFAULT_OBSTACLE_DENSITY))));
        data.put("routeAlgorithm", task.getOrDefault("routeAlgorithm", "BFS"));
        data.put("active", false);                            // ⑳+27 不激活任务
        log.info("SET_CONFIG → 转发到 TaskConfigurator");
        taskActive = false;
        bb.clearTaskQueue();                                  // 清除旧任务
        sendCommand(CommandType.FORWARD_CONFIG, data,
                ConfigConstants.QUEUE_TASK_CONFIG_CMD);
    }

    /**
     * 处理 RESET 任务：用户点了"重置"
     *
     * 强制重建一切：
     *   - forceReset=true → TaskConfigurator 执行 clearAll + 重建
     *   - 停止录制 + 删除旧快照
     *   - 重置 tickCount
     */
    private void handleResetTask(Map<String, String> task) {
        JSONObject data = new JSONObject();
        // ... 解析参数（同 SET_CONFIG）...
        data.put("mapWidth", Integer.parseInt(
                task.getOrDefault("mapWidth",
                        String.valueOf(ConfigConstants.DEFAULT_MAP_WIDTH))));
        data.put("mapHeight", Integer.parseInt(
                task.getOrDefault("mapHeight",
                        String.valueOf(ConfigConstants.DEFAULT_MAP_HEIGHT))));
        data.put("carCount", Integer.parseInt(
                task.getOrDefault("carCount",
                        String.valueOf(Math.max(4, bb.getAllCarIds().size())))));
        data.put("obstacleDensity", Double.parseDouble(
                task.getOrDefault("obstacleDensity",
                        String.valueOf(ConfigConstants.DEFAULT_OBSTACLE_DENSITY))));
        data.put("reset", true);
        data.put("forceReset", true);                         // ⑳+28 ← 关键！强制重建
        data.put("active", false);
        log.info("RESET → 转发到 TaskConfigurator (forceReset=true)");
        userActivated = false;
        taskActive = false;
        recording = false;                                    // 停止录制
        try {
            try (redis.clients.jedis.Jedis j = bb.getJedis()) {
                j.del("replay:snapshots");                    // ⑳+29 清除旧快照
            }
        } catch (Exception e) { /* ignore */ }
        tickCount = 0;                                        // 重置 tick 计数
        bb.clearTaskQueue();
        sendCommand(CommandType.FORWARD_CONFIG, data,
                ConfigConstants.QUEUE_TASK_CONFIG_CMD);
    }

    // ==================== 周期性广播（100ms） ====================

    /**
     * 每 100ms 执行一次的周期性任务
     *
     * 职责（按顺序）：
     *   1. 探索完成判定 → 如果完成则停任务
     *   2. 兜底受阻检查 → 检查所有 BLOCKED 状态的车
     *   3. 驱动空闲小车 → 给 IDLE 且有路径的车发 MOVE_STEP
     *   4. 递增 tick 计数
     *   5. 录制快照
     *   6. 广播 REFRESH_ALL → 刷新所有前端
     */
    private void broadcastTick() {
        if (!running) return;                                // ⑳+30 已停止 → 不执行

        try {
            if (taskActive) {
                // ——— 1. 探索完成判定（双保险） ———
                long unexploredCount = getUnexploredCount();   // SCARD unexplored:set
                double explored = getExploredPercent();        // BITCOUNT 计算探索率
                if (unexploredCount == 0 && explored >= 99.9) {
                    // ⑳+31 两个条件都满足 → 确认完成
                    log.info("巡检完成！未探索格子=0, 探索率={}%", explored);
                    taskActive = false;
                    bb.setTaskActive(false);                   // 写入 Redis
                    recording = false;                         // 自动停止录制
                    wakeTaskProcessor();                       // 唤醒（taskProcessLoop 会进入等待）
                }

                // ——— 2. 兜底受阻检查 ———
                fallbackBlockedCheck();                        // 检查所有 BLOCKED 车，超时的恢复

                // ——— 3. 驱动空闲小车 ———
                // 全局暂停时跳过 tickDrive，但仍继续广播（让前端看到暂停状态）
                if (!bb.isGlobalPaused()) {
                    tickDriveCars();                           // ⑳+32 给 IDLE 且有路径的车发 MOVE_STEP
                }
                tickCount++;                                   // ⑳+33 递增 tick 计数

                // ——— 4. 录制快照 ———
                saveSnapshotIfRecording();

                // 每 20 tick（2 秒）打印一次状态
                if (tickCount % 20 == 0) {
                    log.info("节拍 #{} 完成，未探索剩余: {}, 探索率: {}%",
                            tickCount, unexploredCount, explored);
                }
            }
            // ——— 5. 无论是否 taskActive 都广播 ———
            broadcastViewRefresh();                            // ⑳+34 即使暂停也广播（前端需要看到状态）
        } catch (Exception e) {
            log.error("广播节拍出错", e);
        }
    }

    // ==================== 受阻处理 ====================

    /**
     * 受阻超时恢复
     *
     * 当一辆车被障碍物堵住超过 BLOCKED_TIMEOUT_TICKS(2) 个 tick 后：
     *   1. 获取该车的分布式锁（防止多 Controller 实例同时恢复）
     *   2. 清空该车的当前路径
     *   3. 清空该车的当前目标
     *   4. 设为 IDLE 状态
     *   5. 推入 ROUTE_NEEDED 任务（触发重新规划）
     *
     * @param carId 小车 ID
     */
    private void handleBlockedTimeout(String carId) {
        long blockedTick;
        try {
            blockedTick = bb.getCarBlockedTick(carId);        // 读取受阻时记录的 tick
        } catch (Exception e) {
            blockedTick = 0;                                  // 读失败默认 0
        }
        long diff = tickCount - blockedTick;                  // 计算已经过去了多少个 tick

        if (diff >= ConfigConstants.BLOCKED_TIMEOUT_TICKS) {  // ⑳+35 2 tick = 200ms
            log.info("小车 {} 已阻塞 {} 个节拍，清空路径/目标，转为 IDLE", carId, diff);

            // 获取分布式锁（防止多个 Controller 实例同时恢复）
            DistributedLock lock = bb.getCarLock(carId);
            if (!lock.tryLock()) {
                log.warn("获取小车 {} 锁失败，延迟处理", carId);
                return;                                      // ⑳+36 锁被占用 → 下次 tick 再试
            }
            try {
                bb.clearCarRoute(carId);                     // ⑳+37 清空旧路径（可能指向障碍物）
                bb.clearCarTarget(carId);                    // ⑳+38 清空旧目标（可能已不可达）
                bb.setCarStatus(carId, CarStatus.IDLE);      // ⑳+39 恢复 IDLE
                bb.pushTask("ROUTE_NEEDED", carId, null);    // ⑳+40 触发重新规划
            } finally {
                lock.unlock();                               // ⑳+41 无论如何释放锁
            }
        }
    }

    /**
     * 兜底检查：遍历所有 BLOCKED 状态的车，超时的恢复
     *
     * 为什么需要这个兜底？
     *   - 正常情况下 Car 推入 BLOCKED 任务 → Controller 立刻处理 → handleBlockedTimeout
     *   - 但如果 Car 推入任务的时候 Controller 正好忙于其他事，
     *     或者 BLOCKED 任务丢了（极少见但理论上可能），
     *     fallbackBlockedCheck 作为"安全网"确保不会漏掉
     */
    private void fallbackBlockedCheck() {
        List<String> carIds = getAllCarIds();
        for (String carId : carIds) {
            if (!isMyCar(carIds, carId)) continue;           // ⑳+42 不是我管的车 → 跳过
            try {
                if (bb.getCarStatus(carId) == CarStatus.BLOCKED) {
                    handleBlockedTimeout(carId);
                }
            } catch (Exception e) {
                log.warn("兜底检查车辆 {} 失败", carId, e);
            }
        }
    }

    /**
     * 判断一辆车是否属于当前 Controller 实例
     *
     * 分片规则：carId 在 cars 列表中的索引 % totalInstances == instanceId
     *
     * 示例：totalInstances=2
     *   索引 0 (Car001): 0%2=0 → 实例 0 ✓, 实例 1 ✗
     *   索引 1 (Car002): 1%2=1 → 实例 0 ✗, 实例 1 ✓
     *
     * @param carIds 所有小车 ID 列表
     * @param carId  要检查的小车 ID
     * @return true=归本实例管
     */
    private boolean isMyCar(List<String> carIds, String carId) {
        if (totalInstances <= 1) return true;                // ⑳+43 单实例 → 全部归我管
        int idx = carIds.indexOf(carId);
        return idx >= 0 && (idx % totalInstances) == instanceId;  // ⑳+44 取模匹配
    }

    /**
     * 驱动所有 IDLE 且有路径的小车走一步
     *
     * 每 100ms 执行一次，遍历所有归本实例管的车：
     *   如果车是 IDLE 状态且有待走路径 → 发 MOVE_STEP
     *
     * 这个方法是 tickDriveCars 和 processTask(MOVE_READY) 的"第二入口"
     * MOVE_READY 处理新路径的"第一步"，tickDriveCars 处理后续步骤
     */
    private void tickDriveCars() {
        boolean globalPaused = bb.isGlobalPaused();          // ⑳+45 一次读取复用（减少 Redis 调用）

        List<String> carIds = getAllCarIds();
        for (String carId : carIds) {
            if (!isMyCar(carIds, carId)) continue;           // ⑳+46 不是我管的 → 跳过
            try {
                if (bb.getCarStatus(carId) != CarStatus.IDLE) continue;  // ⑳+47 非空闲 → 跳过

                // ——— 暂停检查 ———
                if (globalPaused) continue;                  // 全局暂停
                String owner = bb.getCarOwner(carId);
                if (owner != null && !owner.isEmpty()
                        && bb.isOperatorPaused(owner)) {      // 个人暂停
                    continue;
                }

                // 检查是否有路径可走
                Point next = bb.peekNextStep(carId);         // ⑳+48 LRANGE -1 -1（只看不取）
                if (next == null) continue;                   // 没有路径

                log.debug("tickDrive: carId={}, next=({},{})", carId, next.x, next.y);
                JSONObject moveData = new JSONObject();
                moveData.put("carId", carId);
                sendCommand(CommandType.MOVE_STEP, moveData,
                        ConfigConstants.carQueueName(carId));  // ⑳+49 发 MOVE_STEP 命令
            } catch (Exception e) {
                log.warn("tickDrive 车辆 {} 失败", carId, e);
                // ⑳+50 单辆车出错不影响其他车
            }
        }
    }

    // ==================== 命令发送 ====================

    /** 发送导航请求：NAVIGATE → Navigator4CarID */
    private void requestNavigate(String carId) {
        JSONObject data = new JSONObject();
        data.put("carId", carId);
        sendCommand(CommandType.NAVIGATE, data,
                ConfigConstants.QUEUE_NAVIGATOR_4_CAR_ID);   // ⑳+51 发给 Navigator 共享队列
    }

    /**
     * 广播刷新信号：REFRESH_ALL → UpdateView Fanout Exchange
     *
     * 只有 instanceId=0 的实例执行广播（避免多个 Controller 重复发送）
     */
    private void broadcastViewRefresh() {
        if (instanceId != 0) return;                         // ⑳+52 只有主实例广播
        JSONObject data = new JSONObject();
        data.put("tick", tickCount);
        sendCommand(CommandType.REFRESH_ALL, data,
                ConfigConstants.EXCHANGE_UPDATE_VIEW, true);  // ⑳+53 fanout=true →广播
    }

    /** 简化版命令发送（非 broadcast） */
    private void sendCommand(CommandType cmd, JSONObject data, String destination) {
        sendCommand(cmd, data, destination, false);
    }

    /**
     * 发送命令的底层实现
     *
     * @param cmd         命令类型（如 NAVIGATE, MOVE_STEP）
     * @param data        命令携带的数据
     * @param destination 目标（队列名或 Exchange 名）
     * @param fanout      true=广播到 Fanout Exchange, false=发送到普通队列
     */
    private void sendCommand(CommandType cmd, JSONObject data,
                              String destination, boolean fanout) {
        JSONObject msg = new JSONObject();
        msg.put("cmd", cmd.name());                           // ⑳+54 枚举名 → 字符串 "NAVIGATE"
        msg.put("data", data);
        msg.put("timestamp", System.currentTimeMillis());
        try {
            if (fanout) {
                mq.fanoutPublish(destination, msg.toJSONString());  // ⑳+55 Fanout 广播
            } else {
                mq.publish(destination, msg.toJSONString());        // ⑳+56 普通队列发送
            }
        } catch (Exception e) {
            log.error("发送消息失败: {} -> {}", cmd, destination, e);
        }
    }

    // ==================== 快照录制 ====================

    /**
     * 录制当前状态的完整快照（如果 recording=true）
     *
     * 每 100ms 一帧，内容包含：
     *   - 所有小车状态（位置、目标、步数、归属）
     *   - 探索 bitmap（1600 字符 "0"/"1" 字符串）
     *   - 障碍物 bitmap（1600 字符）
     *   - 全局暂停状态、探索完成状态
     *
     * 录制失败不影响主流程（catch 中忽略异常）
     */
    private void saveSnapshotIfRecording() {
        if (!recording) return;
        try {
            com.alibaba.fastjson2.JSONObject snap =
                    new com.alibaba.fastjson2.JSONObject();
            snap.put("tick", tickCount);
            snap.put("timestamp", System.currentTimeMillis());
            snap.put("mapWidth", bb.getMapWidth());
            snap.put("mapHeight", bb.getMapHeight());
            snap.put("exploredRate", getExploredPercent() / 100.0);  // ⑳+57 转为 0.0~1.0
            snap.put("taskActive", taskActive);

            // 采集所有小车状态
            var cars = new com.alibaba.fastjson2.JSONArray();
            for (String id : getAllCarIds()) {
                var cs = new com.alibaba.fastjson2.JSONObject();
                cs.put("carId", id);
                cs.put("status", bb.getCarStatus(id).name());
                var p = bb.getCarPosition(id);
                cs.put("position", p != null ? com.alibaba.fastjson2.JSON.toJSON(p) : null);
                cs.put("steps", bb.getCarSteps(id));
                var t = bb.getCarTarget(id);
                cs.put("target", t != null ? com.alibaba.fastjson2.JSON.toJSON(t) : null);
                cs.put("owner", bb.getCarOwner(id));
                cars.add(cs);
            }
            snap.put("cars", cars);

            // 采集探索 bitmap（1600 个 "0"/"1" 字符）
            var map = bb.getMapView();
            var sb = new StringBuilder();
            for (int y = 0; y < bb.getMapHeight(); y++)
                for (int x = 0; x < bb.getMapWidth(); x++)
                    sb.append(map[y][x] ? '1' : '0');
            snap.put("mapBits", sb.toString());

            // 采集障碍物 bitmap
            var blocked = bb.getMapBlocked();
            var sb2 = new StringBuilder();
            for (int y = 0; y < bb.getMapHeight(); y++)
                for (int x = 0; x < bb.getMapWidth(); x++)
                    sb2.append(blocked[y][x] ? '1' : '0');
            snap.put("blockedBits", sb2.toString());

            snap.put("globalPaused", bb.isGlobalPaused());

            // 探索完成状态（双保险判定）
            long unexploredCount = getUnexploredCount();
            double explored = getExploredPercent();
            snap.put("completed", unexploredCount == 0 && explored >= 99.9);

            // RPUSH 追加到快照 List 尾部
            try (redis.clients.jedis.Jedis j = bb.getJedis()) {
                j.rpush("replay:snapshots", snap.toJSONString());
            }
        } catch (Exception e) {
            // ⑳+58 录制失败不影响主流程（系统继续正常运行）
        }
    }

    // ==================== 查询辅助 ====================

    /** 获取探索率（带异常保护） */
    private double getExploredPercent() {
        try { return bb.getExploredPercent(); }
        catch (Exception e) { return 0.0; }
    }

    /** 获取所有小车 ID（带异常保护） */
    private List<String> getAllCarIds() {
        try { return bb.getAllCarIds(); }
        catch (Exception e) {
            // ⑳+59 兜底：如果 Redis 连不上，返回默认 4 辆车
            return Arrays.asList("Car001", "Car002", "Car003", "Car004");
        }
    }

    /** 获取路径算法名（带异常保护） */
    private String getRouteAlgorithm() {
        try { return bb.getRouteAlgorithm(); }
        catch (Exception e) { return "BFS"; }
    }

    /**
     * 获取未探索计数（带异常保护）
     *
     * 异常时返回 Long.MAX_VALUE → broadcastTick 中的判定条件
     * unexploredCount == 0 永远不会成立 → 不会误判为完成
     */
    private long getUnexploredCount() {
        try { return bb.getUnexploredCount(); }
        catch (Exception e) { return Long.MAX_VALUE; }      // ⑳+60 永不误判！
    }

} // ==================== ControllerAgent 结束 ====================
```
